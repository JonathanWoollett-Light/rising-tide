# Rising Tide - AI strategy for the Mirelurk Tribe (MLT)
#
# Every block is allowed only for original_tag = MLT (the Kingdom of M'lyeh is a cosmetic tag, so the tag stays MLT). OWB
# replace_paths common/ai_strategy, so this file only exists when Rising Tide loads after OWB (the descriptor's dependencies).
# Values of one type add up across every active block, OWB's generic ones included (vanilla _documentation.md:158-162, :202).
#
# The conquest order is CONQUEST_PLAN.txt's (repo root): CCW, TRL, RBT; DIS through its schism offer when it holds
# DIS_mend_schism, else by force; MDT; the Kingdom; the Book states' owners (PMR, and ARR while it stands apart from the
# NCR); the Oregon ring; Cascadia; the route to the NCR - the whole north first, whoever holds it by then; New Reno when it
# stands between MLT and the NCR's Nevada allies; the NCR; then the Legion and the interior; then Texas, Mexico and the north.
# Conquests no stage schedules - the north's land in other hands, and countries that cut MLT off from an enemy - are in
# mltd_MLT_frontier.txt.
#
# A. always on: unit mix, research, political power, war discipline, army size, every front, Old Castro, Book networks
# B. one conquest block per target, war-goal holds, the NCR's side   C. defence of The Warren   D. cult operations (La Resistance)

# Unit mix. OWB gives MLT role_ratio infantry +50 -13, marines +15 and amphibious_beast +60 (default.txt:163-204,
# creature_defaults.txt:13-28). A role's share is a base of 100 plus the values (vanilla _documentation.md:158-160), so
# cancelling OWB's generic infantry and marines takes -137 and -115; build_army -999 also stops those two and OWB's motorised
# and mechanised roles outright (OWB's own idiom, _caps_div_limiter.txt and default.txt:571-606) - run 2 ended with generic
# Infantry, Motorized Infantry and Special Ops templates beside our roles. OWB's amphibious_beast keeps a share: its
# divisions are mirelurks. Creature archetypes are type = infantry (cre_amphibious_eq.txt:6), so OWB's
# equipment_production_factor infantry -15 also cut mirelurk production; +15 undoes it. unverified:
# equipment_variant_production_factor on a modded archetype (OWB's only use is anti_air_equipment, default.txt:323-327).
mltd_ai_unit_mix = {
	allowed = { original_tag = MLT }
	enable = { always = yes }
	abort = { always = no }

	ai_strategy = { type = role_ratio id = infantry value = -137 }
	ai_strategy = { type = role_ratio id = marines value = -115 }
	ai_strategy = { type = build_army id = infantry value = -999 }
	ai_strategy = { type = build_army id = marines value = -999 }
	ai_strategy = { type = build_army id = motorized value = -999 }
	ai_strategy = { type = build_army id = mechanized value = -999 }
	ai_strategy = { type = role_ratio id = amphibious_beast value = -30 }
	ai_strategy = { type = role_ratio id = mltd_coral_host value = 60 }
	ai_strategy = { type = equipment_production_factor id = infantry value = 15 }

# ---- A2. The army: cover the borders, then upgrade ------------------------------------------------------------------------
# mltd_ai_survey (on_weekly_MLT) sets mltd_ai_army_target: two divisions a border state, three at war, at least 12, and
# mltd_ai_army_cap a quarter above it. Below the target the AI wants more divisions; above the cap it wants fewer, trains no
# more of its line roles and spends army experience on its templates instead (the upgrades). The Deep Ones and Star Spawn, capped by special_forces_min, keep training. unverified:
# ai_wanted_divisions_factor's unit - read as a percentage change, as HUN.txt:788-791 (240) and CHI.txt:1133-1151 use it.
mltd_ai_army_short = {
	allowed = { original_tag = MLT }
	enable = {
		has_variable = mltd_ai_army_target
		check_variable = { num_divisions < mltd_ai_army_target }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = ai_wanted_divisions_factor value = 50 }
}

mltd_ai_army_full = {

# ---- B. Conquest ------------------------------------------------------------------------------------------------------------
# Each block's stage is the scripted trigger mltd_ai_stage_<tag> (common/scripted_triggers/mltd_ai_triggers.txt).
# One block per target, the shape of OWB's TRL.txt:50-77, CES.txt:110-148 and BRA.txt:198-229. The target must exist, not be
# our subject and not share our faction. A war already under way with it switches the block on; so does its stage, while it
# is a target MLT can take on (mltd_ai_safe_target: not on the NCR's side before the NCR's stage, no ally at half MLT's
# strength); so does the frontier (mltd_ai_frontier_target: it holds land of the north, or stands between MLT and an enemy
# MLT cannot reach), as for the countries in mltd_MLT_frontier.txt.
# - conquer: what makes the AI justify war goals (wiki; unverified in files - no strategy is verified to force a justification).
# - antagonize: lowers the tension bar for a war goal (GENERATE_WARGOAL_ANTAGONIZE_SCALE, vanilla 00_defines.lua:2629).
# - consider_weak: large, because OWB's DECLARE_WAR_RELATIVE_FORCE_FACTOR = 500 (01_defines.lua:292) lets relative force
#   dominate the decision to declare.
# - declare_war: only acts once a war goal exists - a justification, or a conflict capstone's create_wargoal.
# - prepare_for_war: moves troops to the border; the primary target of each stage only.
# - front_unit_request: takes a country_trigger, not an id (TRL.txt:65-71).

# ---- B1. Oregon, 2275: CCW, then TRL, then RBT (CONQUEST_PLAN.txt, 2275) --------------------------------------------------
# CCW from day 0: TRL's Wizard war (~day 133) can only target CCW, and finds nothing to take if CCW's land is already ours.
mltd_ai_conquer_ccw = {
	allowed = { original_tag = MLT }
	enable = {
		country_exists = CCW
		CCW = { NOT = { is_subject_of = ROOT } }
		NOT = { is_in_faction_with = CCW }
		OR = {
			has_war_with = CCW
			AND = {
				mltd_ai_stage_ccw = yes
				CCW = { mltd_ai_safe_target = yes }
			}
			CCW = { mltd_ai_frontier_target = yes }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = conquer id = "CCW" value = 400 }
	ai_strategy = { type = antagonize id = "CCW" value = 200 }
	ai_strategy = { type = consider_weak id = "CCW" value = 500 }
	ai_strategy = { type = prepare_for_war id = "CCW" value = 500 }
	ai_strategy = { type = declare_war id = "CCW" value = 300 }
	ai_strategy = { type = front_unit_request country_trigger = { original_tag = CCW } value = 200 }
}

# ---- D. Cult operations (La Resistance) -------------------------------------------------------------------------------------
# mltd_ai_cult_target is written weekly in on_weekly_MLT from a scorer. The var: guard copies OWB's run_generic_operation
# (generic_operation_strats.txt:1-33); 950 / 940 outrank its generic operation (900) but not its operative rescue (1000).
# One operation per block, each switched on by that operation's own condition on the target, as vanilla does
# (ENG_operation_strats.txt:44-73): Found a Cult where there is no cult, Nurture the Cult where one is under 100 %.
mltd_ai_cult_found = {
	allowed = { original_tag = MLT }
	enable = {
		has_intelligence_agency = yes
		has_completed_focus = mltd_the_spreading_cult
		has_variable = mltd_ai_cult_target
		NOT = { check_variable = { mltd_ai_cult_target = 0 } }
		var:mltd_ai_cult_target = {
			exists = yes
			has_capitulated = no
			NOT = { is_ally_with = PREV }
			NOT = { has_variable = mltd_cult_strength }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = operative_operation operation = mltd_op_found_a_cult operation_target = var:mltd_ai_cult_target value = 950 }
	ai_strategy = { type = operative_mission mission = build_intel_network mission_target = var:mltd_ai_cult_target value = 850 }
	ai_strategy = { type = operation_equipment_priority value = 100 }
}

		var:mltd_ai_cult_target = {
			exists = yes
