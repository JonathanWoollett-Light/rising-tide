# Rising Tide - Cult Infiltration (round 17). A cult's strength lives on the country that holds it, as mltd_cult_strength
# (0-100, shown as a %). The category opens with the focus mltd_the_spreading_cult, which also plants the first cults.
# Cults are founded and grown by La Resistance operations (common/operations/mltd_operations.txt: mltd_op_found_a_cult,
# mltd_op_nurture_the_cult). These two decisions are untargeted and draw on every cult at once through
# mltd_cult_total_strength, the sum of every cult's strength in points, kept on MLT by mltd_cult_rebuild_list (daily).
# Call for People pays 10 manpower per point, Call for Equipment 4 infantry equipment + 1 support equipment per point; each
# costs 25 PP on a 90-day timed-flag cooldown (the house idiom, [?flag:days_left|0]). The rates appear in the temp-variable
# lines below and in mltd_cult_call_*_tt; the cooldown in the flag and in mltd_cult_call_*_again_tt.
mltd_cult_infiltration_cat = {

	mltd_cult_call_for_people = {
		icon = GFX_decision_mltd_cult_hood
		allowed = {
			original_tag = MLT
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = mltd_cult_any_tt
				check_variable = { mltd_cult_total_strength > 0 }
			}
			custom_trigger_tooltip = {
				tooltip = mltd_cult_call_people_cooldown_tt
				NOT = { has_country_flag = mltd_cult_call_people_cooldown }
			}
		}
		cost = 25
		fire_only_once = no
		ai_will_do = {
			factor = 5
			modifier = {
				factor = 0
				check_variable = { mltd_cult_total_strength < 50 }
			}
			# manpower is rarely what the tribe lacks: only when the pool runs low
			modifier = {
				factor = 0
				has_manpower > 20000
			}
		}
		complete_effect = {
			set_temp_variable = { mltd_cult_call_amount = mltd_cult_total_strength }
			multiply_temp_variable = { mltd_cult_call_amount = 10 }
			custom_effect_tooltip = mltd_cult_call_people_tt
			custom_effect_tooltip = mltd_cult_call_people_again_tt
			hidden_effect = {
				add_manpower = mltd_cult_call_amount

			}
		}
	}
