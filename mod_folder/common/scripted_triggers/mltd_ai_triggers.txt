# The NCR's side: the NCR, a member of its faction, its subject, or a country it guarantees - a war on any of them is a war on
# the NCR. Run 2 declared on Arroyo in September 2279, three years before its plan, and fought the NCR's whole faction for six.
# Country scope.
mltd_ai_in_ncr_bloc = {
	country_exists = NCR
	OR = {
		tag = NCR
		is_in_faction_with = NCR
		is_subject_of = NCR
		NCR = { has_guaranteed = PREV }
	}
}

# The NCR, or the Legion, beaten: gone, or capitulated. A capitulated country stays at war while any ally holds out, so
# waiting for it to be gone can wait for ever (run 2's NCR capitulated in September 2283 and still existed in 2285).
# Round 27: also once it has capitulated to MLT at all (the flag on_capitulation records on MLT, mltd_on_actions.txt) or
# is MLT's subject - has_capitulated clears at the peace, so a rump that outlives it would never count again. These are
# the gates of the story acts' closes (When Shady Sands Falls, When the Legion Breaks, as mltd_texas_beaten is Act VI's),
# so the stages that read them (B9, B10), the holds on early war goals and the late army open exactly when the act that
# follows does. MLT is named by tag, so any scope works.
mltd_ai_ncr_beaten = {
	OR = {
		NOT = { country_exists = NCR }
		NCR = { has_capitulated = yes }
		NCR = { is_subject_of = MLT }
		MLT = { has_country_flag = mltd_ncr_capitulated }
	}
}

mltd_ai_legion_beaten = {
	OR = {
		NOT = { country_exists = CES }
		CES = { has_capitulated = yes }
		CES = { is_subject_of = MLT }
		MLT = { has_country_flag = mltd_ces_capitulated }
	}
}

# A country MLT can attack without starting a war it cannot win, as a player would judge it: on the NCR's side only once the
# NCR's own stage is open (then that whole side is the target), and otherwise with no ally - faction member, overlord or
# subject - at half MLT's estimated army strength or more (strength_ratio, vanilla ai_strategy/ENG.txt:1368), and, after the
# Kingdom, no stronger than MLT. mltd_ai_safe_target below adds the courtships; Drown the Dance's hold (section W) reads this
# half alone, because the courtship refuses the Bone Dancers outright. Country scope.
mltd_ai_safe_by_strength = {
	if = {
		limit = { mltd_ai_in_ncr_bloc = yes }
		ROOT = { mltd_ai_stage_ncr = yes }
	}
	else = {
		NOT = {
			any_allied_country = {
				strength_ratio = { tag = ROOT ratio > 0.5 }
			}
		}
		# nor, after the Kingdom, a country stronger than MLT itself (run 3 declared on Broken Coast, 47 states, and lost);
		# the Oregon opening is exempt, because mirelurk armour beats Oregon's weapons in a way estimated strength misses
		OR = {
			ROOT = { NOT = { has_completed_focus = mlt_kingdom_of_mlyeh } }
			NOT = { strength_ratio = { tag = ROOT ratio > 1 } }
		}
	}
}

# A safe target (mltd_ai_safe_by_strength), and never the Broken Coast or the Bone Dancers while MLT courts them. Country
# scope.
mltd_ai_safe_target = {
	# never the Broken Coast or the Bone Dancers while MLT courts them
	NOT = {
		AND = {
			tag = BRK
			ROOT = { mltd_ai_courting_brk = yes }
		}
	}
	NOT = {
		AND = {
			tag = BDT
			ROOT = { mltd_ai_courting_bdt = yes }
		}
	}
	mltd_ai_safe_by_strength = yes
}

# An enemy of MLT that MLT cannot reach by land: it holds land, has not capitulated, and none of it touches land MLT controls
# (is_neighbor_of compares controlled territory, vanilla triggers_documentation.md). Run 2's war never ended because Vault
# City, Elko and S'Lanter held eleven states that touched only New Reno, EAS, BDT and Ouroboros. Country scope (the enemy).
mltd_ai_enemy_cut_off = {
	has_capitulated = no
	num_of_controlled_states > 0
	NOT = { is_neighbor_of = ROOT }
}

# A country at peace with MLT that stands between MLT and such an enemy: it borders both. Country scope.
mltd_ai_cuts_off_an_enemy = {
	NOT = { has_war_with = ROOT }
	is_neighbor_of = ROOT
	any_neighbor_country = {
		has_war_with = ROOT
		mltd_ai_enemy_cut_off = yes
	}
}

# A conquest the stages do not schedule but the map calls for, against a safe target:
# - it holds land of the north once the Kingdom stands (never while the Final Ritual runs): the plan's northern targets die
#   to others - run 2 lost KLA and TDN to MOD (then New Reno), TIM to BDT, and KAM and BRK to SYN - and the north comes first;
# - or it stands between MLT and an enemy MLT cannot reach, MLT has had such an enemy for 90 days (mltd_ai_pocket, set by
#   mltd_ai_survey) and still has no military access through it (the requests in common/ai_strategy/mltd_MLT_frontier.txt).
# Country scope (the candidate).
mltd_ai_frontier_target = {
	OR = {
		AND = {
			ROOT = {
				has_completed_focus = mlt_kingdom_of_mlyeh
				NOT = { has_idea = mltd_the_final_ritual }
			}
			any_owned_state = { mltd_ai_north_state = yes }
		}
		AND = {
			ROOT = { has_country_flag = { flag = mltd_ai_pocket days > 90 } }
			mltd_ai_cuts_off_an_enemy = yes
			NOT = { ROOT = { has_military_access_to = PREV } }
		}
	}
	mltd_ai_safe_target = yes
}

# The NCR's cult is ready for the war: at 40 % or more - a war triples a cult's decay (mltd_cult_weekly), so it has to stand
# before the war - or there is no cult to wait for: no La Resistance, no NCR, or it has had until 2283. MLT scope.
mltd_ai_ncr_cult_ready = {
	OR = {
		NOT = { has_dlc = "La Resistance" }
		NOT = { country_exists = NCR }
		NCR = {
			has_variable = mltd_cult_strength
			check_variable = { mltd_cult_strength > 39 }
		}
		date > 2283.1.1
	}
}

			country_exists = PAL
			PAL = { mltd_ai_in_ncr_bloc = yes }

			SLN = { mltd_ai_in_ncr_bloc = yes }
		}
	}
}

# May an AI MLT take a claim focus (Expanding M'lulu's Domain, Cascadian Current, Sea of Cortez, Tropic of Cancer)?
# They replace OWB's ai_has_no_other_wars_or_wargoals (wargoal_triggers.txt:37-49) in our focus override, which lets an AI
# take them only at peace with no war goal held - a handicap a human never has, and one that kept run 2's MLT from its
# claims for the two years its NCR war could not end. Narrower than removing it: while MLT is not losing (surrender
# progress at 15 %, mltd_ai_no_new_wars_when_losing's line), holds no war goal against a country it is not yet at war with,
# and has no enemy it can still fight - every enemy has capitulated or is cut off (none touches land MLT controls). A
# human always passes, and the hidden_trigger keeps the tooltip as OWB's was. MLT scope.
mltd_ai_may_press_claims = {
	hidden_trigger = {
		if = {
			limit = { is_ai = yes }
			NOT = { surrender_progress > 0.15 }
			NOT = {
				any_other_country = {
					NOT = { has_war_with = ROOT }
					ROOT = { has_wargoal_against = PREV }
				}
			}
			NOT = {
				any_enemy_country = {
					has_capitulated = no
					is_neighbor_of = ROOT
				}
			}
		}
	}
}

# MLT can pay a marketplace lot and keep 150 caps (OWB's own AI floor): caps_cost_trigger on caps_diff, as the player's
# button checks it (mltd_ai_buy_line_equipment). MLT scope.
mltd_ai_org_can_pay = {
	caps_cost_trigger = yes
	set_temp_variable = { mltd_org_left = caps_number_display }
	add_to_temp_variable = { mltd_org_left = caps_diff }
	check_variable = { mltd_org_left > 149 }
}

# Courting the Broken Coast (round 21): from the Kingdom on (the Broken Coast's branch, mltd_brk_* in Act II, opens only
# after The Deep Ones Walk, and the frontier or the Broken Coast's stage would open a war on it before then), while BRK is
# not at war with MLT, is not yet in its faction and has not refused the Drowned Covenant (global flag, event mltd.21), MLT
# helps BRK win its coast instead of conquering it. MLT scope.
mltd_ai_courting_brk = {
	has_completed_focus = mlt_kingdom_of_mlyeh
	country_exists = BRK
	NOT = { has_war_with = BRK }
	NOT = { is_in_faction_with = BRK }
	NOT = { has_global_flag = mltd_brk_refused_covenant }
}

# Broken Coast helped before the NCR (round 21): once The Deep Ones Sail North (Act II, taken before The Final Ritual) has
# raised the volunteer limit, the AI sends the Broken Coast volunteers before it turns on the NCR - at least four divisions
# there for 26 weeks (mltd_ai_brk_volunteer_weeks, counted by the weekly survey; volunteers go home while MLT is at war, so
# this is peace-time help) - or it is no longer needed: no courtship, Broken Coast at peace, capitulated or holding its
# coast, the Covenant (Act III) taken, or 2283. MLT scope.
mltd_ai_brk_helped = {
	OR = {
		NOT = { mltd_ai_courting_brk = yes }
		BRK = {
			OR = {
				has_war = no
				has_capitulated = yes
				has_completed_focus = brk_nf_pirate_coast
			}
		}
		has_completed_focus = mltd_brk_the_drowned_covenant
		check_variable = { mltd_ai_brk_volunteer_weeks > 25 }
		date > 2283.1.1
	}
}

# Courting the Bone Dancers (round 22): while OWB's Odious King branch is still open to them, they have not refused the
# Covenant, are not at war with MLT and MLT has not chosen Drown the Dance instead, the AI takes the invitation (Hail the
# Drowned King, Act III) and leaves them alone. Gated on the Kingdom rather than on mltd_bdt_the_bone_shore (round 24): The
# Bone Shore (Act II) opens only after The Deep Ones Walk, and until it was taken the frontier could conquer BDT and close
# the Covenant for good - mltd_ai_courting_brk gates on the Kingdom for exactly this reason. MLT scope.
mltd_ai_courting_bdt = {
	has_completed_focus = mlt_kingdom_of_mlyeh
	country_exists = BDT
	# The invitation needs the Bone Dancers on OWB's Odious King branch: either they have crowned him, or they still can.
	# bdt_hail_to_the_king is mutually exclusive with bdt_road_warriors_of_the_eighty_four and bdt_listen_to_the_pilgrims
	# (OWB Bone Dancers (BDT) Focus.txt:1508-1513), so either of those closes the Covenant for good and frees the war branch.
	BDT = {
		OR = {
			has_completed_focus = bdt_hail_to_the_king
			AND = {
				NOT = { has_completed_focus = bdt_road_warriors_of_the_eighty_four }
				NOT = { has_completed_focus = bdt_listen_to_the_pilgrims }
			}
		}
	}
	NOT = { has_completed_focus = mltd_bdt_drown_the_dance }
	NOT = { has_war_with = BDT }
	NOT = { is_in_faction_with = BDT }
	NOT = { has_global_flag = mltd_bdt_refused_covenant }
}

# ---- W. The story acts' war focuses: when an AI may take one -------------------------------------------------------------
# Round 27 review. Each war focus declares its own war on completion (common/national_focus/mltd_act3..6_focus.txt), so its
# 42-49 days are the run-up. An AI MLT takes every focus of its strategy plan's ai_national_focuses that is AVAILABLE, in list
# order, whatever its ai_will_do reads: run 8 (ai_runs/20260918-170847, round-24 code) completed Eyes in the Sound on day
# 1330 and The Tide-Wall on day 1715 with mltd_ai_stage_wbh closed (20 states, not 31), A Mole in Shady Sands on day 1750
# before The Deep Ones Walk and The Turbines Sing on day 2513 with the NCR's stage never open - each while its ai_will_do
# read 0. (Its two Muttfruit FOCUS lines are no evidence either way: both trades were bypassed - by owning 186, and 493 and
# 450 - and a bypassed focus reads as completed.) So each war focus calls its trigger below in `available`: hidden, and for
# an AI only (a human always passes), the idiom of mltd_ai_may_press_claims. Each holds only while the focus would actually
# start a war: with its target gone, already at war with MLT or MLT's subject, the focus is a payout and nothing waits. This
# only restricts the AI, so it is fair play. MLT scope (ROOT is MLT).
#
# Every hold also waits while MLT is losing a war (surrender progress over 15 %, the line of mltd_ai_may_press_claims and
# mltd_ai_no_new_wars_when_losing - which only damps the strategy's own declarations and cannot reach a focus's). It lifts as
# soon as MLT is not losing, so it strands no act. The optional wars - their act's close also opens through a sibling - wait
# for a safe target as well (mltd_ai_safe_target, or mltd_ai_safe_by_strength for the courted Bone Dancers), as the conquest
# blocks for the same targets demand, because a focus's own declaration skips that check. Act III's two wars are an AI's real
# ways through the act (an AI seldom meets both invitations' gates), so their safe-target test gives way on 1 January 2281,
# the date The Tide Turns South's own size gate gives way. The Turbines Sing, The Sea Against Mars and Black Water Rising,
# whose closes need their wars, wait on no safe target.
# A hold is read only when the AI picks the focus: every war focus is continue_if_invalid = yes, so one started while MLT was
# winning still declares on completion if MLT has started losing during the run-up.

# The Tide-Wall (Act III): Washington's stage (more than 30 states, the Kingdom and the Walk), the army ready, MLT not losing
# and - until 2281 - every Brotherhood it would declare on a safe target, while a Brotherhood that holds Washington stands at
# peace with MLT and is not its subject.
mltd_ai_may_raise_the_tide_wall = {
	hidden_trigger = {
		if = {
			limit = {
				is_ai = yes
				any_country = {
					mltd_wbh_brotherhood = yes
					NOT = { has_war_with = ROOT }
					NOT = { is_subject_of = ROOT }
				}
			}
			mltd_ai_stage_wbh = yes
			mltd_ai_army_ready = yes
			NOT = { surrender_progress > 0.15 }
			OR = {
				NOT = {
					any_country = {
						mltd_wbh_brotherhood = yes
						NOT = { has_war_with = ROOT }
						NOT = { is_subject_of = ROOT }
						NOT = { mltd_ai_safe_target = yes }
					}
				}
				date > 2281.1.1
			}
		}
	}
}

# Drown the Dance (Act III): the army ready, MLT not losing, and not while MLT courts the Bone Dancers (mltd_ai_courting_bdt:
# Hail the Drowned King is the plan's answer to them) - until 1 January 2281. An AI builds their network, takes their civilian
# token (mltd_ai_network_bdt, mltd_ai_infiltrate_bdt) and points its cult operations at them, but whether Hail's gates open in
# time rests on the Bone Dancers' own tree; from 2281 it stops waiting, so that Act III's close has a Bone Dancers path besides
# The Tide-Wall's. Until 2281 also only against Bone Dancers with no strong ally and no strength to beat MLT
# (mltd_ai_safe_by_strength: mltd_ai_safe_target itself refuses them while they are courted, which would cancel the date).
# Never while MLT fights an NCR that has not capitulated: the courtship's date would otherwise open a Bone Dancers front in the
# middle of the NCR's war, which on the plan's timeline runs through 2281.
mltd_ai_may_drown_the_dance = {
	hidden_trigger = {
		if = {
			limit = {
				is_ai = yes
				country_exists = BDT
				NOT = { has_war_with = BDT }
				BDT = { NOT = { is_subject_of = ROOT } }
			}
			mltd_ai_army_ready = yes
			NOT = { surrender_progress > 0.15 }
			OR = {
				NOT = { mltd_ai_courting_bdt = yes }
				date > 2281.1.1
			}
			OR = {
				BDT = { mltd_ai_safe_by_strength = yes }
				date > 2281.1.1
			}
			NOT = {
				AND = {
					has_war_with = NCR
					NCR = { has_capitulated = no }
				}
			}
		}
	}
}

# The Turbines Sing (Act IV): the NCR's stage (the north consolidated, the NCR's cult at 40 %, the Broken Coast helped, New
# Reno out of the way - or 2284), the army ready and MLT not losing. Lifted only once the focus can no longer start the war
# early: the NCR gone, MLT's subject or already at war with MLT, or once it has capitulated to MLT (mltd_ncr_capitulated, set
# by on_capitulation) - Act IV's close needs the focus, so an AI must not be held off it after MLT's war. A capitulation to
# anyone else (the Legion in the Hoover war, say) does not lift it: the NCR stays capitulated for as long as its allies fight
# on, and the focus would still declare on it - and on its whole side - with the stage closed.
mltd_ai_may_start_ncr_war = {
	hidden_trigger = {
		if = {
			limit = {
				is_ai = yes
				country_exists = NCR
				NOT = { has_war_with = NCR }
				NCR = { NOT = { is_subject_of = ROOT } }
				NOT = { has_country_flag = mltd_ncr_capitulated }
			}
			mltd_ai_stage_ncr = yes
			mltd_ai_army_ready = yes
			NOT = { surrender_progress > 0.15 }
		}
	}
}

# mltd_ai_conquer_ate demands (round 21: no war on a stronger country or one with a strong ally), because the focus's own
# declaration skips that check; and never while a Texan war the close waits on still runs (a capitulated Texan tag counts as
# beaten, as in mltd_texas_beaten).
mltd_ai_may_start_ate_war = {
	hidden_trigger = {
		if = {
			limit = {
				is_ai = yes
				country_exists = ATE
				NOT = { has_war_with = ATE }
				ATE = { NOT = { is_subject_of = ROOT } }
			}
			mltd_ai_stage_ate = yes
			mltd_ai_army_ready = yes
			NOT = { surrender_progress > 0.15 }
			ATE = { mltd_ai_safe_target = yes }
			NOT = {
				AND = {
					has_war_with = TBH
					TBH = { has_capitulated = no }
				}
			}
			NOT = {
				AND = {
					has_war_with = LNS
					LNS = { has_capitulated = no }
				}
			}
		}
	}
}
