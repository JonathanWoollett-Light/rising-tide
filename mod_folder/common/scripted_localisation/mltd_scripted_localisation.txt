# GetMltdWmRequirement: the requirement line of The Wet Market's rows, where OWB's rows show "Minimum Opinion": green once
# the Mirelurk Tribe has completed the focus The Wet Market, red before.
defined_text = {
	name = GetMltdWmRequirement
	text = {
		trigger = {
			has_global_flag = mltd_wet_market_open
		}
		localization_key = mltd_wm_requirement_open
	}
	text = {
		localization_key = mltd_wm_requirement_closed
	}
}

# GetMltdWmMarketOpen / GetMltdWmAvailability: The Wet Market tab's tooltip, in the shape of OWB's seller tabs
# (organization_ledger_opinion_tt). "Market Open" says whether the Mirelurk Tribe has completed the focus The Wet Market
# (the global flag it sets), "Available for transaction" whether our 40-day trade cooldown has run out. The yes branches
# reuse OWB's own org_available ("§GYes§!").
defined_text = {
	name = GetMltdWmMarketOpen
	text = {
		trigger = {
			has_global_flag = mltd_wet_market_open
		}
		localization_key = org_available
	}
	text = {
		localization_key = mltd_wm_market_closed
	}
}
defined_text = {
	name = GetMltdWmAvailability
	text = {
		trigger = {
			NOT = { has_country_flag = mltd_wet_market_recent_trade }
		}
		localization_key = org_available
	}
	text = {
		localization_key = mltd_wm_unavailable
	}
}

# GetMltd<Brk|Bdt|Wbh>Odds (round 28): how likely an invitation's answer is, at the end of the invitation's description -
# The Drowned Covenant (mltd.21), Hail the Drowned King (mltd.24) and Terms for the Citadel (mltd.32). Each reads the game
# rule, so it is true in the game being played: the answer is those events' ai_chance, and a historical focus rule changes it
# (is_historical_focus_on, as OWB's nevada_pact_events.txt:15-22 weights its own). Change the odds and these lines together.
# A human on the other side answers as they please, so each first says so; the Broken Coast's also covers raiders already
# gone or MLT's subject (the focus then pays political power), and the Brotherhood's a Washington no free Brotherhood holds.
# Terms for the Citadel asks WBH unless it is gone or someone's subject, else TCA as the Northwestern Brotherhood
# (mltd_wbh_terms_target; OWB lets TCA form it while WBH is its subject), so the Brotherhood's lines follow the same rule: a
# subject Brotherhood is never the one that answers. Tags are written out (never ROOT): a focus description names no scope
# of its own.
defined_text = {
	name = GetMltdBrkOdds
	text = {
		trigger = {
			OR = {
				NOT = { country_exists = BRK }
				BRK = { is_subject_of = MLT }
			}
		}
		localization_key = mltd_brk_odds_gone
	}
	text = {
		trigger = {
			BRK = { is_ai = no }
		}
		localization_key = mltd_brk_odds_human
	}
	text = {
		trigger = {
			is_historical_focus_on = yes
		}
		localization_key = mltd_brk_odds_historical
	}
	text = {
		localization_key = mltd_brk_odds_unhistorical
	}
}
defined_text = {
	name = GetMltdBdtOdds
	text = {
		trigger = {
			country_exists = BDT
			BDT = { is_ai = no }
		}
		localization_key = mltd_bdt_odds_human
	}
	text = {
		localization_key = mltd_bdt_odds
	}
}
