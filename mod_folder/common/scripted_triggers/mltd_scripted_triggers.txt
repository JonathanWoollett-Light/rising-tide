# Rising Tide - MLT scripted triggers
#
# mltd_flavour_can_fire: shared gate for the flavour events mltd.9-19, drawn each month
# from the random_events pool in on_monthly_MLT (common/on_actions/mltd_on_actions.txt). Open once the
# Kingdom of M'lyeh is proclaimed; shut while capitulated and while mltd_flavour_cooldown runs (120 days,
# set in each flavour event's immediate block; event mltd.1 also sets it for 90 days).
mltd_flavour_can_fire = {
	original_tag = MLT
	has_completed_focus = mlt_kingdom_of_mlyeh
	has_capitulated = no
	NOT = { has_country_flag = mltd_flavour_cooldown }
}

# mltd_hail_candidate: where event mltd.19 (Hail M'lyeh) can find its convert. Scope: the candidate country, ROOT = MLT.
# MLT has a network there - national coverage above 0, OWB's own gate for operation_infiltrate_civilian
# (00_operations.txt:135-138) - and does not already hold that country's civilian infiltration token (the event's reward).
mltd_hail_candidate = {
	exists = yes
	NOT = { tag = ROOT }
	ROOT = {
		network_national_coverage = {
			target = PREV
			value > 0
		}
		NOT = {
			has_operation_token = {
				tag = PREV
				token = token_civilian
			}
		}
	}
}

# mltd_cult_neighbouring_major: a major that neighbours ROOT (MLT), used by the focus
# mltd_the_spreading_cult (every_country) and by mltd_cult_can_be_founded (the decision). Plain is_major,
# by design. OWB also makes every non-subject country a major while it is at war (00_on_actions.txt:
# 1362-1370, flag set_as_major_for_war; taken away at peace, :431-446), so a small neighbour at war counts
# for as long as the war lasts; at peace only the top industrial powers are majors (fchanges_defines.lua:
# MIN_MAJOR_COUNTRIES = 3, MAJOR_MIN_FACTORIES = 40).
mltd_cult_neighbouring_major = {
	exists = yes
	NOT = { tag = ROOT }
	is_neighbor_of = ROOT
	NOT = { is_subject_of = ROOT }
	is_major = yes
}

# mltd_cult_can_be_founded: FROM scope of the decision mltd_cult_found (ROOT = MLT) - a neighbouring
# major without a cult yet. Defined after mltd_cult_neighbouring_major, which it calls.
mltd_cult_can_be_founded = {
	mltd_cult_neighbouring_major = yes
	NOT = { has_variable = mltd_cult_strength }
}

# mltd_defector_candidate: a general of the target of mltd_op_indoctrinate_the_faithful_* who may defect to MLT on its
# critical success. Scope: the character, inside FROM's any_army_leader / random_army_leader (which skip leaders not yet
# visible). set_nationality carries every role along, so anyone who leads or could lead the target, advises it or is an
# exile it shelters stays; so do field marshals (the operation takes "one of their generals") and OWB's story generals.
mltd_defector_candidate = {
	is_field_marshal = no
	is_country_leader = no
	can_be_country_leader = no
	is_advisor = no
	is_exiled_leader = no
	mltd_defector_story_character = no
}

# mltd_wet_market_can_buy: the buy conditions of every Wet Market item (scripted GUI mltd_wet_market_panel), each an
# OWB-style tooltip line. Reads the temp variable mltd_wm_price (caps) set by the button's trigger. The first line is the
# market itself: the Mirelurk Tribe completed The Wet Market (the focus sets the global flag).
mltd_wet_market_can_buy = {
	custom_trigger_tooltip = {
		tooltip = mltd_wet_market_open_tt
		has_global_flag = mltd_wet_market_open
	}
	set_temp_variable = { caps_diff = mltd_wm_price }
	multiply_temp_variable = { caps_diff = -1 }
	caps_cost_trigger = yes
	custom_trigger_tooltip = {
		tooltip = mltd_wet_market_stock_tt
		check_variable = { global.mltd_wet_market_stock > 249 }
	}
	custom_trigger_tooltip = {
		tooltip = mltd_wet_market_cooldown_tt
		NOT = { has_country_flag = mltd_wet_market_recent_trade }
	}
}

# mltd_wet_market_can_sell: the sell conditions shared by every Wet Market row (the button's own trigger adds "at least
# 250 of that variant"). The market must be open, and selling counts against the same 40-day cooldown as buying.
mltd_wet_market_can_sell = {
	custom_trigger_tooltip = {
		tooltip = mltd_wet_market_open_tt
		has_global_flag = mltd_wet_market_open
	}
	custom_trigger_tooltip = {
		tooltip = mltd_wet_market_cooldown_tt
		NOT = { has_country_flag = mltd_wet_market_recent_trade }
	}
}
