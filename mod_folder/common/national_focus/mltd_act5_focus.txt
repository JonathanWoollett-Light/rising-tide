# Rising Tide (MLT) - Act V, "The Sea Against Mars" (round 27, the story acts).
# Shared focuses, pulled into mlt_nf through their prerequisites: the chapter, mltd_mars_in_the_water, hangs off Act IV's
# close, mltd_when_shady_sands_falls (mltd_act4_focus.txt), which reaches the listed Act III roots through Act IV's chapter
# and mltd_the_tide_turns_south; everything else here hangs off the chapter. No focus in this file is listed in the override.
#
# The act folds four of the old conflict sub-trees (round 13) into one chapter of the spine (x = 15):
#   row 31  the chapter   Mars in the Water (0,1) of mltd_when_shady_sands_falls - event mltd.50, the river runs salt
#   row 32  the fan       What the River Carries (-3,1), The Drowned Paladin (-1,1), The Lake God Answers (1,1),
#                         Whispers in the Steam (3,1)
#   row 33  the wars      The Sea Against Mars (-3,2), Blood in the Water (-1,2), Rites on the Spiral Jetty (1,2),
#                         The Crusade Turns South (3,2) - each under its own head, each declaring its war
#   row 34  the close     When the Legion Breaks (0,3) - any one war opens it (one OR block), once Caesar's Legion has
#                         capitulated (to anyone now, or to us in a war since ended: the flag mltd_ces_capitulated), is gone
#                         or is our subject; the idea mltd_mars_beneath_the_waves, event mltd.51 (the pens: the same 5,000
#                         people either freed into the capital with +5 % stability, or fed to the pools as 5,000 manpower)
#                         and world news mltd.52
#
# Merged away, each into the kept focus that now carries its distinctive effects (pure duplicates dropped - a second
# intelligence grant to the same country is folded into one add_intel at the larger values):
#   The Drowned Frumentarius           -> mltd_ces_what_the_river_carries (Silanus the Drowned, CES token_army, CES -6 % stability)
#   A Crate for Every Cross            -> mltd_ces_the_sea_against_mars (800 infantry and 200 support equipment, or 100 caps)
#   The Chained Are Given to the Tide  -> event mltd.51, option a (the spec's +5,000 capital population, +5 % stability)
#   The Paladin Returns                -> mltd_bos_the_drowned_paladin (BOS token_army, BOS -35 political power)
#   The Codex Unsealed                 -> mltd_bos_the_drowned_paladin (the stolen tech bonus, BOS -4 % stability)
#   The Tide Takes the Armoury         -> mltd_bos_blood_in_the_water (the energy-weapon and power-armour theft, its tiers)
#   The Drowned Shamans                -> mltd_wht_the_lake_god_answers (+40 political power, WHT -6 % stability, the converts)
#   Guns for Both Sides                -> mltd_wht_the_lake_god_answers (the per-buyer sale, round 24's fix)
#   Sand in the Engines                -> mltd_wht_rites_on_the_spiral_jetty (EHT -3 % war support, 600 infantry equipment)
#   A Second Schism                    -> mltd_hea_whispers_in_the_steam (+40 political power, HEA -35 political power and
#                                         -6 % stability, HEA token_army)
#   Steam for the Deep                 -> mltd_hea_the_crusade_turns_south (600 infantry equipment, the stolen industry bonus)
#
# What else changed (rules in the round-27 build spec): no head has a Kingdom, country or intelligence gate any more (the
# intel rewards stay, inside has_dlc = "La Resistance"), and each pays a fallback when its target is gone; the round-23
# is_ai modifiers that waited on the NCR's stage are gone, because the tree now enforces the order. The four wars keep their
# gates (the Utah road war or 2280, Heaven's Guard's holy wars or 2280) and their AI readiness hold (mltd_ai_army_ready; since
# the round-27 review in `available`, mltd_ai_may_start_<ces|bos|utah|hea>_war, because an AI takes any listed focus that is
# available whatever its ai_will_do reads); their round-22 stage term, mltd_ai_ncr_beaten, is dropped for the same reason (the
# chapter needs Act IV's close), since it could only strand an AI once a beaten NCR outlived the peace. Since the review each
# hold also waits while MLT is losing a war, and Blood in the Water, the Rites and the Crusade - optional, since the close
# opens through The Sea Against Mars - wait for every target they would declare on to be a safe target. Kept focuses keep their
# ids, costs and icons; tech bonuses that were named after a deleted focus are now named after the focus that grants them.
# Every focus is cancel_if_invalid = no + continue_if_invalid = yes.

# The cult spies on Flagstaff and turns a frumentarius (merges The Drowned Frumentarius)
shared_focus = {
	id = mltd_ces_what_the_river_carries
	icon = GFX_goal_EXE_something_fishy
	cost = 35
	prerequisite = { focus = mltd_mars_in_the_water }
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = -3
	y = 1
	relative_position_id = mltd_mars_in_the_water
	ai_will_do = {
		factor = 3
	}
	completion_reward = {
		mltd_intel_foothold = yes
		# round 17: the cults of M'lyeh in this branch's countries wither half as fast (mltd_cult_refresh)
		if = {
			limit = { country_exists = CES }
			custom_effect_tooltip = mltd_ces_cult_sheltered_tt
			hidden_effect = {
				CES = { set_country_flag = mltd_cult_sheltered }
			}
		}
		if = {
			limit = { has_dlc = "La Resistance" }
			# the drowned frumentarius (merged): an operative once the agency exists, as before
			if = {
				limit = { has_intelligence_agency = yes }
				create_operative_leader = {
					name = mltd_ces_drowned_frumentarius_operative
					gfx = GFX_Portrait_CES_Operator_3
					skill = 2
					traits = { operative_frumentarius }
					bypass_recruitment = no
					available_to_spy_master = yes
					nationalities = { MLT CES }
				}
			}
			else = {
				add_political_power = 50
			}
			# one intelligence grant (the river's civ 10 / army 10 and the frumentarius' civ 15, at the larger values) and the
			# frumentarius' army token; the Legion gone, the river's +25 stands in for both
			if = {
				limit = { country_exists = CES }
				add_operation_token = {
					tag = CES
					token = token_army
				}
				add_intel = {
					target = CES
					civilian_intel = 15
					army_intel = 10
				}
			}
			else = {
				add_political_power = 25
			}
		}
		else = {
			# 25 (What the River Carries) + 50 (The Drowned Frumentarius), their two no-DLC fallbacks
			add_political_power = 75
		}
		if = {
			limit = { country_exists = CES }
			CES = { add_stability = -0.06 }
		}
	}
}

# The war on the Legion (merges A Crate for Every Cross)
shared_focus = {
	id = mltd_ces_the_sea_against_mars
	icon = GFX_goal_CES_Mars_God
	cost = 45
	prerequisite = { focus = mltd_ces_what_the_river_carries }
	available = {
		# round 22: the focus is the declaration, so an AI takes it only when the army is ready - only while it would start a
		# war. Round 27: its stage term (mltd_ai_ncr_beaten) is gone - the chapter already needs Act IV's close, and the term
		# could only hold the AI back once a beaten NCR outlived the peace (has_capitulated clears then), stranding Act V, Act VI
		# and the finale. Since the round-27 review the hold sits here, hidden and AI-only, because an AI takes any listed focus
		# that is available whatever its ai_will_do reads; it also waits while MLT is losing a war (mltd_ai_triggers.txt,
		# section W)
		mltd_ai_may_start_ces_war = yes
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = -3
	y = 2
	relative_position_id = mltd_mars_in_the_water
	ai_will_do = {
		factor = 3
	}
	will_lead_to_war_with = CES
	completion_reward = {
		# round 22: the branch ends in the war itself - the focus declares on each living target that is not already at war
		# with us or our subject, so its own length is the run-up and will_lead_to_war_with warns them (round 17 granted a war
		# goal the AI could sit on for years)
		if = {
			limit = {
				country_exists = CES
				NOT = { has_war_with = CES }
				CES = { NOT = { is_subject_of = ROOT } }
			}
			declare_war_on = {
				type = annex_everything
				target = CES
			}
		}
		# A Crate for Every Cross (merged): the frumentarius empties the armouries as the war begins. Paired transfers, the
		# victim's removal with no producer (CLAUDE.md > Theft without producer); guards at least the amount taken.
		if = {
			limit = {
				country_exists = CES
				CES = { has_equipment = { infantry_equipment > 1599 } }
			}
			CES = {
				add_equipment_to_stockpile = {
					type = infantry_equipment
					amount = -800
				}
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = 800
				producer = CES
			}
		}
		else = {
			set_temp_variable = { caps_to_add = 100 }
			add_caps = yes
		}
		if = {
			limit = {
				country_exists = CES
				CES = { has_equipment = { support_equipment > 399 } }
			}
			CES = {
				add_equipment_to_stockpile = {
					type = support_equipment
					amount = -200
				}
			}
			add_equipment_to_stockpile = {
				type = support_equipment
				amount = 200
				producer = CES
			}
		}
		add_timed_idea = { idea = mltd_ces_the_drowned_bull days = 180 }
		# OWB's own Legion scripting: Caesar's illness (CES_stress, nf_legion.26 from CES's first focus), his death (nf_legion.70 retires CES_caesar), the succession war (CES_civil_war)
		if = {
			limit = { NOT = { country_exists = CES } }
			custom_effect_tooltip = mltd_ces_mars_fallen_tt
			add_political_power = 50
		}
		else_if = {
			limit = {
				country_exists = CES
				CES = {
					OR = {
						has_country_flag = CES_civil_war
						NOT = { has_country_leader = { character = CES_caesar } }
					}
				}
			}
			custom_effect_tooltip = mltd_ces_mars_dead_tt
			army_experience = 40
			CES = { add_stability = -0.06 }
		}
		else_if = {
			limit = {
				country_exists = CES
				CES = { has_country_flag = CES_stress }
			}
			custom_effect_tooltip = mltd_ces_mars_fever_tt
			set_temp_variable = { caps_to_add = 150 }
			add_caps = yes
			CES = { add_war_support = -0.03 }
		}
		else = {
			custom_effect_tooltip = mltd_ces_mars_hale_tt
			add_political_power = 50
		}
		if = {
			limit = {
				country_exists = CES
				CES = {
					OR = {
						has_idea = cultofmars
						has_idea = cultofmars_principes
						has_idea = cultofmars_triplex
					}
				}
			}
			custom_effect_tooltip = mltd_ces_mars_cult_tt
			CES = { add_war_support = -0.02 }
		}
	}
}
