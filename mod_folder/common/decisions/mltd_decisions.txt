# Rising Tide - Cult Infiltration (targeted decisions: FROM is the target country).
# A cult's strength lives on the country that holds it, as mltd_cult_strength (0-100, shown as a %).
# The category and all five decisions open with the focus mltd_the_spreading_cult.
# Founding needs a neighbouring major (mltd_cult_can_be_founded, common/scripted_triggers); the other
# four need a living cult there, and three of them need La Resistance intelligence on the target.
# Costs are paid in hidden_effect, as OWB's custom_cost decisions do (area51_decisions.txt): each literal
# appears in custom_cost_trigger (PP and manpower as N-1, caps as caps_diff = -N), in complete_effect and in
# the mltd_cult_cost_<id> loc text and its _blocked twin. With the caps rule off, add_caps charges caps as
# 1.25 x political power (caps_scripted_effects.txt:89-92) and caps_cost_trigger always passes for a cost,
# so the trigger's else gates on PP + 1.25 x caps (as N-1) instead.
# Cooldowns are countdown variables on the target, in weeks, lowered by mltd_cult_weekly (on_weekly_MLT);
# each literal appears in the set_variable and in mltd_cult_<id>_again_tt. Gains appear twice (the tooltip's
# temp variable and the one inside FROM). Mechanics: common/scripted_effects/mltd_scripted_effects.txt.
mltd_cult_infiltration_cat = {

	mltd_cult_found = {
		icon = GFX_decision_mltd_cult_hood
		allowed = {
			original_tag = MLT
		}
		target_array = neighbors
		target_root_trigger = {
			has_completed_focus = mltd_the_spreading_cult
		}
		target_trigger = {
			FROM = {
				mltd_cult_can_be_founded = yes
			}
		}
		custom_cost_trigger = {
			has_manpower > 999
			if = {
				limit = { has_global_flag = caps_enabled_global_flag }
				has_political_power > 49
				set_temp_variable = { caps_diff = -50 }
				caps_cost_trigger = yes
			}
			else = {
				has_political_power > 111.5	# 50 PP + 50 caps, which add_caps takes as 62.5 PP
			}
		}
		custom_cost_text = mltd_cult_cost_found
		fire_only_once = no
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				has_political_power < 150
			}
		}
		complete_effect = {
			set_temp_variable = { mltd_cult_gain = 10 }
			custom_effect_tooltip = mltd_cult_found_tt
			hidden_effect = {
				add_political_power = -50
				add_manpower = -1000
				set_temp_variable = { caps_to_add = -50 }
				add_caps = yes
				FROM = {
					set_temp_variable = { mltd_cult_gain = 10 }

				}
