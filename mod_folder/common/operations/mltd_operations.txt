# MLTD OPERATIONS #
# Rising Tide - Mirelurk Tribe (MLT). The Coral Court sends its oracles abroad to gather the faithful for M'lulu. #
# Reuses existing OWB phases only (each has empty requirements and equipment); no new phase, token or gfx files are required. #
#
# Indoctrinate the Faithful comes as five variants, one per band of the target's manpower: an operation's equipment is a
# fixed base cost (vanilla common/operations/_documentation.md:151-158), so a cost that follows the target needs one
# definition per band. Each variant is visible only while the target's manpower is in its band, so the intelligence
# screen shows exactly one; they share their name, description, icon and phases. The lowest also shows, greyed, at
# 1,000 manpower or less. Manpower moved and support equipment:
#   > 50,000: 5,000 / 500    > 25,000: 2,500 / 250    > 10,000: 1,000 / 100    > 5,000: 250 / 25    > 1,000: 100 / 10
# At completion mltd_indoctrinate_transfer moves the variant's amount, or less if the target has since fallen into a lower
# band, so MLT never gains manpower the target does not lose. cost_multiplier = 0 turns off the engine's default +15 % cost
# per repeat against the same target (DEFAULT_OPERATION_COST_MULTIPLIER, vanilla 00_defines.lua:3733-3737), so the
# equipment is exactly the figure above every time. The critical success (outcome_extra_execute replaces the base
# outcome, so it repeats the transfer) also turns one of the target's generals: mltd_indoctrinate_defector.
#
# Found a Cult and Nurture the Cult (round 17, at the end of the file) replace the targeted cult decisions.

	priority = 10
	days = 60
	network_strength = 30
	operatives = 2

	allowed = {
		original_tag = MLT
	}

	visible = {
		ROOT = {
			original_tag = MLT
		}
		network_national_coverage = {
			target = FROM
			value > 0
		}
		FROM = {
			has_manpower > 50000
		}
	}

	available = {
		FROM = {
			has_manpower > 50000
		}
	}

	equipment = {
		support_equipment = 500
	}
	cost_multiplier = 0

	experience = 2.5

	risk_chance = 0.1
	outcome_extra_chance = 0.25
	risk_modifiers = { operation_infiltrate_risk operation_risk }
	outcome_modifiers = { operation_infiltrate_outcome operation_outcome }

	outcome_execute = {
		ROOT = {
			set_temp_variable = { mltd_op_cap = 5000 }
			mltd_indoctrinate_transfer = yes

		}
	}
	outcome_potential = {
		FROM = {
			add_manpower = -5000
		}
		ROOT = {
			add_manpower = 5000
		}
		custom_effect_tooltip = mltd_op_indoctrinate_the_faithful_tt
	}

	ai_will_do = {
		base = 5
	}

	phases = { #make contact with the disaffected
		infiltrate_military_ideological_supporter = { base = 100 }
	}
	phases = { #gather the faithful
		recruit_mercs = { base = 100 }
	}

	}
}

# ---------------------------------------------------------------- Cult Infiltration (round 17)
# Found a Cult and Nurture the Cult replace round 14's targeted decisions. Both need The Spreading Cult (which opens the
# category) and, like every operation, a network in the target; founding reaches any country, not just neighbouring
# majors. Founding plants a cult at 10 % (20 on a critical success); nurturing adds 12 (20 on a critical success) and is
# paid in infantry and support equipment. The gains are in the outcome blocks and in mltd_op_*_tt; the equipment is
# shown natively. Effects: mltd_cult_add_strength in FROM (the target), then mltd_cult_rebuild_list in ROOT.
mltd_op_found_a_cult = {
	icon = GFX_operations_infiltrate_civilian_govt
	map_icon = GFX_operations_infiltrate_civilian_govt_map
	name = mltd_op_found_a_cult
	desc = mltd_op_found_a_cult_desc

	priority = 10
	days = 45
	network_strength = 20
	operatives = 1

	allowed = {
		original_tag = MLT
	}

	visible = {
		ROOT = {
			original_tag = MLT
			has_completed_focus = mltd_the_spreading_cult
		}
		network_national_coverage = {
			target = FROM
			value > 0
		}
		FROM = {
			NOT = { has_variable = mltd_cult_strength }
		}
	}

	available = {
		FROM = {
			NOT = { has_variable = mltd_cult_strength }
		}
	}

	equipment = {
		infantry_equipment = 100
		support_equipment = 20
	}
	cost_multiplier = 0

	experience = 2

	risk_chance = 0.1
	outcome_extra_chance = 0.25
	risk_modifiers = { operation_infiltrate_risk operation_risk }
	outcome_modifiers = { operation_infiltrate_outcome operation_outcome }
	cost_modifiers = { operation_infiltrate_cost operation_cost }

	outcome_execute = {
		FROM = {
			set_temp_variable = { mltd_cult_gain = 10 }
			mltd_cult_add_strength = yes
		}
		ROOT = {
			mltd_cult_rebuild_list = yes
		}
	}
	outcome_extra_execute = {
		FROM = {
			set_temp_variable = { mltd_cult_gain = 20 }
			mltd_cult_add_strength = yes
		}
		ROOT = {
			mltd_cult_rebuild_list = yes
		}
	}
	outcome_potential = {
		custom_effect_tooltip = mltd_op_found_a_cult_tt
	}

		base = 20
	}

	}

		recruit_mercs = { base = 100 }

	}
}
