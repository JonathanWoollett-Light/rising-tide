# The Final Ritual: every gift becomes permanent. remove_mission skips timeout_effect, so the
# running gifts are not switched off before the variables are pinned to their full values.
# mltd_gift_manpower_var (the modifier's weekly_manpower, set by the ritual focuses) is deliberately
# NOT reset here - it must survive the Final Ritual.
mltd_all_gifts_permanent = {
	custom_effect_tooltip = mltd_all_gifts_permanent_tt
	hidden_effect = {
		if = {
			limit = { has_active_mission = mltd_gift_of_the_tide_mission }
			remove_mission = mltd_gift_of_the_tide_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_shell_mission }
			remove_mission = mltd_gift_of_the_shell_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_current_mission }
			remove_mission = mltd_gift_of_the_current_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_brood_mission }
			remove_mission = mltd_gift_of_the_brood_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_abyss_mission }
			remove_mission = mltd_gift_of_the_abyss_mission
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = mltd_gifts_from_the_deep } } }
			add_dynamic_modifier = { modifier = mltd_gifts_from_the_deep }
		}
		set_variable = { mltd_gift_tide_var = 0.20 }
		set_variable = { mltd_gift_shell_var = 0.20 }
		set_variable = { mltd_gift_current_var = 0.20 }
		set_variable = { mltd_gift_brood_var = 0.30 }
		set_variable = { mltd_gift_abyss_var = 0.20 }
		set_variable = { mltd_gifts_active = 0 }
		set_country_flag = mltd_gifts_permanent
		force_update_dynamic_modifier = yes
	}
}

# the division's need (8 x 20 x 0.2 = 32 Deep Ones, 8 x 10 x 0.2 = 16 Star Spawn), a small reinforcement pool. Until

# National population (country scope), recomputed every day by on_daily_MLT: the sum of
# state_population over every owned state. The ritual focuses gate on it with check_variable
# and print it with [?mltd_national_population|0]. PREV inside the state loop is the country
# (OWB exodus_effects.txt:385 sums state_population_k into PREV.core_population_k the same way).
mltd_update_national_population = {
	set_variable = { mltd_national_population = 0 }
	set_variable = { mltd_controlled_population = 0 }
	every_owned_state = {
		add_to_variable = { PREV.mltd_national_population = state_population }
		# The summons gate on this one, because it is exactly the pool mltd_take_population_everywhere
		# spends from - same is_controlled_by / > 500 filter. Gating on the owned total instead would
		# let an occupied MLT clear the gate and then spread the whole toll over the few states it
		# still holds, or, with nothing controlled, pay nothing at all and get the division free.
		if = {
			limit = {
				is_controlled_by = ROOT
				state_population > 500
			}
			add_to_variable = { PREV.mltd_controlled_population = state_population }
		}
	}
}

# The summon toll. Country scope; set mltd_toll_target_k first, in THOUSANDS of people. Takes that
# many IN TOTAL, split across every owned+controlled state over 500 population in proportion to what
# each holds, so the big states carry most of it and none is emptied - a state with a tenth of the
# nation pays a tenth. Two passes: sum the pool, then charge each state population * target / pool.
# Both variables are country variables read from state scope through PREV, the same way the pool is
# accumulated (as in mltd_update_national_population), so nothing depends on a temp variable
# surviving a scope change. add_manpower in state scope is population, and it is used directly
# rather than through OWB's add_state_population so the state tooltips collapse into the one custom
# line the decision shows.
#
# ORDER AND UNITS MATTER. Script variables are 32-bit fixed point with 3 decimals, so they overflow
# above 2,147,483.647 and WRAP NEGATIVE (OWB names the number at TON_farming_scripted_effects.txt:11
# and detects the wrap at coring_button_scripted_triggers.txt:141). Multiplying population by the
# target first would reach 28,000 x 16,000 = 448,000,000, two hundred times over the ceiling, on
# every single state. Dividing first keeps the intermediate at population/pool_k, which can never
# exceed 1,000, and the pool is summed with state_population_k (vanilla's own overflow-avoiding
# variant, dynamic_variables_documentation.md:1168) so the divisor is in thousands. Peak intermediate
# for MLT's kingdom is 372, about 360x under the ceiling, and the charges still sum to the target
# within one person.
mltd_take_population_everywhere = {
	set_variable = { mltd_toll_pool_k = 0 }
	every_owned_state = {
		limit = {
			is_controlled_by = ROOT
			state_population > 500
		}
		add_to_variable = { PREV.mltd_toll_pool_k = state_population_k }
	}
	if = {
		limit = { check_variable = { mltd_toll_pool_k > 0 } }
		every_owned_state = {
			limit = {
				is_controlled_by = ROOT
				state_population > 500
			}
			set_temp_variable = { mltd_toll_temp = state_population }
			divide_temp_variable = { mltd_toll_temp = PREV.mltd_toll_pool_k }
			multiply_temp_variable = { mltd_toll_temp = PREV.mltd_toll_target_k }
			multiply_temp_variable = { mltd_toll_temp = -1 }
			round_temp_variable = mltd_toll_temp
			# The gate is a variable refreshed once a day, so between refreshes the pool can shrink
			# (states lost, occupied, or drained by the ritual toll). Floor every charge at whatever
			# leaves the state 500 people, so no single state can ever be emptied by a summoning.
			set_temp_variable = { mltd_toll_floor_temp = state_population }
			add_to_temp_variable = { mltd_toll_floor_temp = -500 }
			multiply_temp_variable = { mltd_toll_floor_temp = -1 }
			clamp_temp_variable = { var = mltd_toll_temp min = mltd_toll_floor_temp }
			add_manpower = mltd_toll_temp
		}
	}
}

# The ritual toll (on_daily_MLT while mltd_the_grand_ritual / mltd_the_final_ritual is held).
# Country scope. The factor is copied to a temp variable before the state loop (temp variables
# survive scope changes - OWB exodus_effects.txt sets pop_add then reads it in capital_scope).
# Per state: population * factor, negated, at least 5 (max = -5), rounded, then state-scope
# add_manpower, which is population (tlaloc_on_actions.txt does the same with state_pop).
mltd_ritual_daily_toll = {
	set_temp_variable = { mltd_toll_factor_temp = mltd_ritual_toll_factor }
	every_owned_state = {
		limit = { state_population > 500 }
		set_temp_variable = { mltd_toll_temp = state_population }
		multiply_temp_variable = { mltd_toll_temp = mltd_toll_factor_temp }
		multiply_temp_variable = { mltd_toll_temp = -1 }
		clamp_temp_variable = { var = mltd_toll_temp max = -1 }
		round_temp_variable = mltd_toll_temp
		add_manpower = mltd_toll_temp
	}
}

# Every conflict sub-tree (common/national_focus/mltd_conflicts_focus.txt) opens with this. With
# La Resistance and an agency it adds one random OWB agency upgrade. OWB charges 15 caps in each
# upgrade's complete_effect (intelligence_agency_upgrades.txt:16-34), so the upgrade is refunded, as
# Unbound (UNB) Focus.txt:370-375 does - paid first, so add_caps's bankruptcy check never sees the dip.
# Skipped at OWB's 30-upgrade ceiling, where add_random_agency_upgrade upgrades nothing. It no longer
# founds the agency - the focus mltd_the_spreading_cult does - so without an agency, without the DLC, or at the ceiling
# it pays political power instead.
mltd_intel_foothold = {
	if = {
		limit = {
			has_dlc = "La Resistance"
			has_intelligence_agency = yes
			NOT = { agency_upgrade_number > 29 }
		}
		custom_effect_tooltip = mltd_intel_foothold_upgrade_tt
		hidden_effect = {
			set_temp_variable = { caps_to_add = 15 }
			add_caps = yes
			set_temp_variable = { random_agency_upgrade_amount = 1 }
			add_random_agency_upgrade = yes
		}
	}
	else = {
		add_political_power = 50
	}
}

# ---------------------------------------------------------------- Cult Infiltration
# All four run in the scope of the country that holds (or is about to hold) the cult. MLT is named by
# tag, never ROOT: mltd_cult_refresh also runs from on_war_relation_added, where ROOT can be either side.
# Each is defined before anything that calls it - a console reload logged the forward references
# (mltd_cult_add_strength -> mltd_cult_refresh, mltd_cult_weekly -> mltd_cult_dissolve) as unknown.
# Variables on that country:
#   mltd_cult_strength          0-100, shown as a percentage
#   mltd_cult_decay             points it loses at the next weekly tick
#   mltd_cult_threat_big/_small/_air  the war penalties in percentage points, for loc (0.25 / 0.15 / 0.5 per point)
#   mltd_cult_*_var             the five mltd_cult_infiltration inputs, 0 unless at war with MLT
# Country flag mltd_cult_sheltered (set by the root of each conflict sub-tree on its targets) halves the decay.
# MLT keeps mltd_cult_total_strength, the sum of every cult's strength, which the Call decisions pay out on.

# The last of the faithful are gone: the country is clean, and a new cult can be founded there.
mltd_cult_dissolve = {
	remove_dynamic_modifier = { modifier = mltd_cult_infiltration }
	clear_variable = mltd_cult_strength
	clear_variable = mltd_cult_decay
	clear_variable = mltd_cult_threat_big
	clear_variable = mltd_cult_threat_small
	clear_variable = mltd_cult_threat_air
	clear_variable = mltd_cult_stability_var
	clear_variable = mltd_cult_war_support_var
	clear_variable = mltd_cult_org_var
	clear_variable = mltd_cult_factory_var
	clear_variable = mltd_cult_air_var
}

# Recomputes the weekly decay and the war penalties from the current strength s.
# decay = 0.5 + s * s / 2500 points a week: 1.5 at 50 %, 3.74 at 90 %, 4.5 at 100 %, so holding 90 costs
# ~2.5x what holding 50 does. Halved in a country whose conflict sub-tree MLT has opened (mltd_cult_sheltered); tripled
# while at war with MLT: the faithful rise, and rising exposes them.
# s * s peaks at 10,000, far under the 2,147,483 variable ceiling.
mltd_cult_refresh = {
	set_variable = { mltd_cult_decay = mltd_cult_strength }
	multiply_variable = { mltd_cult_decay = mltd_cult_strength }
	divide_variable = { mltd_cult_decay = 2500 }
	add_to_variable = { mltd_cult_decay = 0.5 }
	if = {
		limit = { has_country_flag = mltd_cult_sheltered }
		multiply_variable = { mltd_cult_decay = 0.5 }
	}
	if = {
		limit = { has_war_with = MLT }
		multiply_variable = { mltd_cult_decay = 3 }
	}
	set_variable = { mltd_cult_threat_big = mltd_cult_strength }
	multiply_variable = { mltd_cult_threat_big = 0.25 }
	set_variable = { mltd_cult_threat_small = mltd_cult_strength }
	multiply_variable = { mltd_cult_threat_small = 0.15 }
	set_variable = { mltd_cult_threat_air = mltd_cult_strength }
	multiply_variable = { mltd_cult_threat_air = 0.5 }
	if = {
		limit = { has_war_with = MLT }
		set_variable = { mltd_cult_stability_var = mltd_cult_threat_big }
		multiply_variable = { mltd_cult_stability_var = -0.01 }
		set_variable = { mltd_cult_war_support_var = mltd_cult_stability_var }
		set_variable = { mltd_cult_org_var = mltd_cult_threat_small }
		multiply_variable = { mltd_cult_org_var = -0.01 }
		set_variable = { mltd_cult_factory_var = mltd_cult_org_var }
		set_variable = { mltd_cult_air_var = mltd_cult_threat_air }
		multiply_variable = { mltd_cult_air_var = 0.01 }
	}
	else = {
		set_variable = { mltd_cult_stability_var = 0 }
		set_variable = { mltd_cult_war_support_var = 0 }
		set_variable = { mltd_cult_org_var = 0 }
		set_variable = { mltd_cult_factory_var = 0 }
		set_variable = { mltd_cult_air_var = 0 }
	}
	force_update_dynamic_modifier = yes
}

# Adds the temp variable mltd_cult_gain to the cult, founding it (and the modifier) first if needed.
mltd_cult_add_strength = {
	if = {
		limit = { NOT = { has_variable = mltd_cult_strength } }
		set_variable = { mltd_cult_strength = 0 }
	}
	add_to_variable = { mltd_cult_strength = mltd_cult_gain }
	clamp_variable = { var = mltd_cult_strength min = 0 max = 100 }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = mltd_cult_infiltration } } }
		add_dynamic_modifier = { modifier = mltd_cult_infiltration }
	}
	mltd_cult_refresh = yes
}

# Weekly (on_weekly_MLT): lose this week's decay, and dissolve below 1 %.
mltd_cult_weekly = {
	mltd_cult_refresh = yes
	subtract_from_variable = { mltd_cult_strength = mltd_cult_decay }
	clamp_variable = { var = mltd_cult_strength min = 0 max = 100 }
	if = {
		limit = { check_variable = { mltd_cult_strength < 1 } }
		mltd_cult_dissolve = yes
	}
	else = {
		mltd_cult_refresh = yes
	}
}

# mltd_cult_rebuild_list: the cults abroad, strongest first, for the Cult Infiltration category's description
# (GetMltdCultList, common/scripted_localisation). MLT scope. Rebuilt rather than kept up piecemeal: every_country skips
# countries that no longer exist, so annexed ones drop out. Called daily by on_daily_MLT once The Spreading Cult is done,
# weekly after the decay loop, by both cult operations and by the focus mltd_the_spreading_cult. It also sums
# mltd_cult_total_strength, which the Call decisions pay out on.
# The loc reads fixed slots on MLT - mltd_cult_row_<k> (the country) and mltd_cult_row_<k>_str (its strength), k = 1-8,
# plus mltd_cult_count and mltd_cult_more - never an array element: stepping into var:array^i from a decision category's
# description found nothing in game. Slots past mltd_cult_count keep stale values that the loc never shows. Selection sort
# with find_highest_in_array and remove_from_array by index (OWB caps_scripted_ai.txt:161-218).
mltd_cult_rebuild_list = {
	clear_array = mltd_cult_list
	clear_array = mltd_cult_sort_country
	clear_array = mltd_cult_sort_strength
	set_variable = { mltd_cult_count = 0 }
	set_variable = { mltd_cult_total_strength = 0 }
	every_country = {
		limit = { has_variable = mltd_cult_strength }
		add_to_array = { ROOT.mltd_cult_sort_country = THIS }
		add_to_array = { ROOT.mltd_cult_sort_strength = mltd_cult_strength }
		add_to_variable = { ROOT.mltd_cult_total_strength = mltd_cult_strength }
	}
	while_loop_effect = {
		limit = { check_variable = { mltd_cult_sort_country^num > 0 } }
		find_highest_in_array = { array = mltd_cult_sort_strength index = mltd_cult_hi }

# mltd_indoctrinate_transfer: the manpower of Indoctrinate the Faithful (mltd_op_indoctrinate_the_faithful_<amount>), ROOT
# (MLT) scope, FROM the target. Moves the variant's amount (temp mltd_op_cap) - or less, if the target's manpower has
# since fallen into a lower band - from FROM to ROOT, so it never takes more than the target holds. Bands (common/operations/
# mltd_operations.txt): > 50,000 5,000, > 25,000 2,500, > 10,000 1,000, > 5,000 250, > 1,000 100; at 1,000 or less, nothing.
mltd_indoctrinate_transfer = {
	if = {
		limit = {
			check_variable = { mltd_op_cap > 4999 }
			FROM = { has_manpower > 50000 }
		}
		FROM = { add_manpower = -5000 }
		add_manpower = 5000
	}
	else_if = {
		limit = {
			check_variable = { mltd_op_cap > 2499 }
			FROM = { has_manpower > 25000 }
		}
		FROM = { add_manpower = -2500 }
		add_manpower = 2500
	}
	else_if = {
		limit = {
			check_variable = { mltd_op_cap > 999 }
			FROM = { has_manpower > 10000 }
		}
		FROM = { add_manpower = -1000 }
		add_manpower = 1000
	}
	else_if = {
		limit = {
			check_variable = { mltd_op_cap > 249 }
			FROM = { has_manpower > 5000 }
		}
		FROM = { add_manpower = -250 }
		add_manpower = 250
	}
	else_if = {
		limit = {
			FROM = { has_manpower > 1000 }
		}
		FROM = { add_manpower = -100 }
		add_manpower = 100
	}
}

# mltd_indoctrinate_defector: the critical success of Indoctrinate the Faithful, ROOT (MLT) scope, FROM the target. One of
# FROM's generals - a mltd_defector_candidate (never a field marshal, a leader or would-be leader, an advisor, an exile or
# a story character), an unassigned one when there is one - is flagged and handed to ROOT with set_nationality (every role,
# skill and trait goes along); event mltd.20 names them. If the one who came is invisible in MLT (a role's visible trigger
# can test the old country), they are sent back. With no valid general, nothing more happens.
mltd_indoctrinate_defector = {
	FROM = {
		hidden_effect = {
			if = {
				limit = {
					any_army_leader = {
						mltd_defector_candidate = yes
						is_assigned = no
					}
				}
				random_army_leader = {
					limit = {
						mltd_defector_candidate = yes
						is_assigned = no
					}
					set_character_flag = mltd_defector_new
					set_nationality = ROOT
				}
			}
			else = {
				random_army_leader = {
					limit = {
						mltd_defector_candidate = yes
					}
					set_character_flag = mltd_defector_new
					set_nationality = ROOT
				}
			}
		}
	}
	if = {
		limit = {
			any_army_leader = {
				has_character_flag = mltd_defector_new
			}
		}
		hidden_effect = {
			set_variable = { mltd_defector_origin = FROM }
			country_event = { id = mltd.20 }
		}
	}
	else = {
		hidden_effect = {
			random_army_leader = {
				include_invisible = yes
				limit = {
					has_character_flag = mltd_defector_new
				}
				clr_character_flag = mltd_defector_new
				set_nationality = FROM
			}
		}
	}
}

				name = mltd_drowned_covenant

			}
		}
		else = {
			create_faction = mltd_drowned_covenant
		}
	}

			is_faction_leader = yes

			}
		}
		add_to_faction = BDT
	}
	add_opinion_modifier = {
		target = BDT
		modifier = mltd_drowned_covenant
	}
	reverse_add_opinion_modifier = {
		target = BDT
		modifier = mltd_drowned_covenant
	}
}
