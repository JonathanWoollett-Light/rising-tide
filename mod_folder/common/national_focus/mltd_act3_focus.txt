# Rising Tide (MLT) - Act III, "The Stars Are Right" (round 27, the story acts).
# After The Final Ritual the north is decided. Four shared focuses fan out of the Final Ritual (a national focus in
# "Mirelurk Tribe (MLT) Focus.txt") and reconverge, through one OR prerequisite, on the act's close, The Tide Turns South:
#
#   row 24                       [The Final Ritual]                       national (override)
#   row 25  [The Tide-Wall] [The Drowned Covenant] [Hail the Drowned King] [Drown the Dance]
#   row 26                    [The Tide Turns South]                      close (new; event mltd.30, news mltd.31)
#
# Pull-in: the four fan focuses name a national focus in their prerequisite, so they reach the tree only because the
# override lists each of them (`shared_focus = <id>`, as vanilla lists CONGO_congo_investments at belgium.txt:20); The Tide
# Turns South comes in through its prerequisites on them. All five take absolute x/y and no relative_position_id.
#
# The four fan focuses are kept from the old conflict sub-trees (rounds 17-24) and do what they did there. What changed:
# - Each now hangs off The Final Ritual instead of its old sub-tree; the old prerequisites are gone. The Covenant and Hail the
#   Drowned King instead name their old sub-tree's root in `available` (mltd_brk_salt_on_the_broken_coast,
#   mltd_bdt_the_bone_shore - Act II heads), which draws no line.
# - Never strand: the close needs any one of the four. Drown the Dance is takeable whenever the Bone Dancers are gone or are
#   MLT's subject (+50 political power instead of the war), or are not MLT's allies - and an independent Bone Dancers can
#   only be MLT's ally through Hail the Drowned King, which then already satisfies the OR. The Covenant is takeable, for +50
#   political power, once the Broken Coast is gone or is MLT's subject. The Tide-Wall pays 30 army experience instead of its
#   penalty and idea when no Brotherhood, or only a subject one, holds Washington. The Covenant and Hail keep their courtship
#   gates (cult, token, coverage, intel): they are invitations.
# - The two wars keep their AI stage, readiness and courtship holds, now only while the focus would actually start a war,
#   so an AI is not held off a fallback it can take for free. Since the round-27 review the holds sit in `available`
#   (mltd_ai_may_raise_the_tide_wall, mltd_ai_may_drown_the_dance - hidden, AI-only), not in ai_will_do: an AI takes any
#   listed focus that is available, whatever its ai_will_do reads. Both also wait while MLT is losing a war and, until
#   2281, for a target without a strong ally or the strength to beat MLT; Drown the Dance never opens a front in the middle
#   of a war with an NCR that has not capitulated.
# Nothing was merged into this act; nothing was deleted here.

# The Tide-Wall of The Warren: war on whichever Brotherhood holds Washington, bunkers in The Warren, its claims gone
shared_focus = {
	id = mltd_wbh_the_tide_wall
	icon = GFX_goal_generic_walls_focus
	cost = 45
	prerequisite = { focus = mltd_the_final_ritual }
	available = {
		custom_trigger_tooltip = {
			tooltip = mltd_wbh_crusade_turns_tt
			OR = {
				NOT = { any_country = { mltd_wbh_brotherhood = yes } }
				any_country = {
					mltd_wbh_brotherhood = yes
					OR = {
						has_completed_focus = WBH_eradicate_the_mutants
						has_war_with = ROOT
					}
				}
				date > 2280.1.1
			}
		}
		235 = { is_owned_and_controlled_by = ROOT }
		# round 22: the focus is the declaration, so an AI takes it only when its stage has come and the army is ready - only
		# while it would start a war (with no Brotherhood left to declare on it pays 30 army experience and nothing waits).
		# Round 27 review: the hold sits here, hidden and AI-only, because an AI takes any listed focus that is available
		# whatever its ai_will_do reads; it also waits while MLT is losing and, until 2281, for every Brotherhood it would
		# declare on to be a safe target (mltd_ai_triggers.txt, section W)
		mltd_ai_may_raise_the_tide_wall = yes
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = 12
	y = 25
	ai_will_do = { factor = 3 }
	will_lead_to_war_with = WBH
	# Round 24: the capstone declares on whichever Brotherhood holds Washington (mltd_wbh_brotherhood), so the warning
	# has to name both - TCA is the branch round 22 was written for.
	will_lead_to_war_with = TCA
	completion_reward = {
		# round 22: the focus declares on each living target that is not already at war with us or our subject, so its own
		# length is the run-up and will_lead_to_war_with warns them
		every_country = {
			limit = {
				mltd_wbh_brotherhood = yes
				NOT = { has_war_with = ROOT }
				NOT = { is_subject_of = ROOT }
			}
			ROOT = {
				declare_war_on = {
					type = annex_everything
					target = PREV
				}
			}
		}
		if = {
			limit = { 235 = { is_owned_and_controlled_by = ROOT } }
			235 = {
				add_building_construction = {
					type = bunker
					level = 2
					instant_build = yes
					province = 2158
				}
				add_building_construction = {
					type = bunker
					level = 1
					instant_build = yes
					province = 2159
				}
				add_building_construction = {
					type = bunker
					level = 1
					instant_build = yes
					province = 2160
				}
				add_building_construction = {
					type = bunker
					level = 1
					instant_build = yes
					province = 2161
				}
			}
		}
		if = {
			limit = { any_country = { mltd_wbh_brotherhood = yes } }
			# the Brotherhood's claim on The Warren: WBH gets it when TRL dies (Shared Brotherhood Focus.txt:2113), TCA when
			# it forms the Northwestern Brotherhood (every Washington state, formable_nations.txt)
			235 = {
				if = {
					limit = { is_claimed_by = WBH }
					remove_claim_by = WBH
				}
				if = {
					limit = { is_claimed_by = TCA }
					remove_claim_by = TCA
				}
			}
		}
		# Round 27: the penalty and the idea only against a Brotherhood that is not MLT's subject; with none left, or only a
		# subject one, the focus pays army experience instead (the spec's rule 5)
		if = {
			limit = {
				any_country = {
					mltd_wbh_brotherhood = yes
					NOT = { is_subject_of = ROOT }
				}
			}
			every_country = {
				limit = {
					mltd_wbh_brotherhood = yes
					NOT = { is_subject_of = ROOT }
				}
				add_war_support = -0.03
			}
			add_timed_idea = {
				idea = mltd_wbh_tide_wall
				days = 180
			}
		}
		else = {
			army_experience = 30
		}
	}
}

# The Drowned Covenant: the Broken Coast is invited into a faction led by MLT (event mltd.21, which BRK may refuse)
shared_focus = {
	id = mltd_brk_the_drowned_covenant
	icon = GFX_goal_BRK_black_flag_fleets
	cost = 49
	prerequisite = { focus = mltd_the_final_ritual }
	available = {
		# the old prerequisite (the sub-tree's root, now an Act II head), as a condition: no line across the tree
		has_completed_focus = mltd_brk_salt_on_the_broken_coast
		# the courtship gates, while there is a Broken Coast to court; once it is gone, or is MLT's own subject, the focus pays
		# political power instead
		if = {
			limit = {
				country_exists = BRK
				BRK = { NOT = { is_subject_of = ROOT } }
			}
			NOT = { has_war_with = BRK }
			BRK = {
				is_subject = no
				is_in_faction = no
			}
			OR = {
				is_in_faction = no
				is_faction_leader = yes
			}
			# the raiders hold the coast their own Dawn of a New Age asks for (brk_nf_pirate_coast: every core of VIC and DRE
			# owned by their sphere), or have already taken that focus
			custom_trigger_tooltip = {
				tooltip = mltd_brk_pirate_coast_tt
				OR = {
					BRK = { has_completed_focus = brk_nf_pirate_coast }
					BRK = {
						VIC = { all_core_land_owned_by_prev_sphere = yes }
						DRE = { all_core_land_owned_by_prev_sphere = yes }
					}
				}
			}
			if = {
				limit = { has_dlc = "La Resistance" }
				custom_trigger_tooltip = {
					tooltip = mltd_brk_strong_cult_tt
					BRK = { check_variable = { mltd_cult_strength > 49.9 } }
				}
				has_operation_token = { tag = BRK token = token_civilian }
				custom_trigger_tooltip = {
					tooltip = mltd_brk_network_tt
					network_national_coverage = { target = BRK value > 0.3 }
				}
				custom_trigger_tooltip = {
					tooltip = mltd_brk_intel_tt
					intel_level_over = {
						target = BRK
						civilian_intel > 0.5
					}
				}
			}
		}
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = 14
	y = 25
	ai_will_do = { factor = 3 }
	completion_reward = {
		if = {
			limit = {
				country_exists = BRK
				BRK = { NOT = { is_subject_of = ROOT } }
			}
			custom_effect_tooltip = mltd_brk_covenant_offer_tt
			custom_effect_tooltip = mltd_newline_tt
			custom_effect_tooltip = mltd_brk_covenant_if_accepted_tt
			hidden_effect = {
				BRK = { country_event = { id = mltd.21 days = 1 } }
			}
		}
		else = {
			add_political_power = 50
		}
	}
}

# Hail the Drowned King: the Bone Dancers, once they have crowned the Odious King, are invited into the Covenant (event mltd.24)
shared_focus = {
	id = mltd_bdt_hail_the_drowned_king
	icon = GFX_goal_BDT_odious_king
	cost = 49
	prerequisite = { focus = mltd_the_final_ritual }
	# one-sided on purpose (round 24): see mltd_bdt_drown_the_dance
	mutually_exclusive = { focus = mltd_bdt_drown_the_dance }
	available = {
		# the old sub-tree's root (now an Act II head), as a condition: no line across the tree
		has_completed_focus = mltd_bdt_the_bone_shore
		country_exists = BDT
		NOT = { has_war_with = BDT }
		BDT = {
			is_subject = no
			is_in_faction = no
			has_completed_focus = bdt_hail_to_the_king
		}
		OR = {
			is_in_faction = no
			is_faction_leader = yes
		}
		if = {
			limit = { has_dlc = "La Resistance" }
			custom_trigger_tooltip = {
				tooltip = mltd_bdt_strong_cult_tt
				BDT = { check_variable = { mltd_cult_strength > 39.9 } }
			}
			has_operation_token = { tag = BDT token = token_civilian }
			custom_trigger_tooltip = {
				tooltip = mltd_bdt_network_tt
				network_national_coverage = { target = BDT value > 0.25 }
			}
		}
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = 16
	y = 25
	# the late plan lists the invitation before Drown the Dance, and its order is what an AI with a plan follows; this 4
	# (over the war's 3) only orders the two for an AI running no plan - a subject MLT, whose plans all abort
	ai_will_do = { factor = 4 }
	completion_reward = {
		custom_effect_tooltip = mltd_bdt_covenant_offer_tt
		custom_effect_tooltip = mltd_newline_tt
		custom_effect_tooltip = mltd_brk_covenant_if_accepted_tt
		hidden_effect = {
			BDT = { country_event = { id = mltd.24 days = 1 } }
		}
	}
}

# Drown the Dance: war on the Bone Dancers (an annex_everything declaration), or political power once they are gone
shared_focus = {
	id = mltd_bdt_drown_the_dance
	icon = GFX_goal_BDT_bloody_skull
	cost = 49
	prerequisite = { focus = mltd_the_final_ritual }
	# Round 24: the exclusion is one-sided on purpose. Drowning the Dance still bars Hail the Drowned King (that focus names
	# this one), but a refused Covenant no longer strands the war branch: mltd_bdt_hail_the_drowned_king completes whatever
	# the Bone Dancers answer, and a human BDT that refuses used to leave MLT with 49 days spent and no option at all. One-sided
	# mutual exclusion is vanilla's own idiom - 742 of its pairs are declared on one side only.
	available = {
		# never on an ally who took the Covenant; with no Bone Dancers left, or with the Bone Dancers MLT's own subject (who
		# sit in its faction if it leads one), the focus is free and pays political power (round 27)
		if = {
			limit = {
				country_exists = BDT
				BDT = { NOT = { is_subject_of = ROOT } }
			}
			NOT = { is_in_faction_with = BDT }
		}
		# round 22: the focus is the declaration, so an AI takes it only with the army ready and not while it courts the Bone
		# Dancers (until 2281) - only while there are Bone Dancers to declare on. Round 27 review: the hold sits here, hidden
		# and AI-only, because an AI takes any listed focus that is available whatever its ai_will_do reads; it also waits
		# while MLT is losing, for a safe target until 2281, and while MLT fights an NCR that has not capitulated
		# (mltd_ai_triggers.txt, section W)
		mltd_ai_may_drown_the_dance = yes
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = 18
	y = 25
	ai_will_do = { factor = 3 }
	will_lead_to_war_with = BDT
	completion_reward = {
		if = {
			limit = {
				country_exists = BDT
				NOT = { has_war_with = BDT }
				BDT = { NOT = { is_subject_of = ROOT } }
			}
			declare_war_on = {
				type = annex_everything
				target = BDT
			}
		}
		# the penalties and the experience only against Bone Dancers that are not MLT's subject; otherwise the fallback
		if = {
			limit = {
				country_exists = BDT
				BDT = { NOT = { is_subject_of = ROOT } }
			}
			BDT = {
				add_stability = -0.06
				add_war_support = -0.03
			}
			army_experience = 40
		}
		else = {
			add_political_power = 50
		}
	}
}

# The Tide Turns South (round 27, Act III's close): the north is the tribe's, and the drums turn toward California
shared_focus = {
	id = mltd_the_tide_turns_south
	icon = GFX_goal_BRK_drums_of_war
	cost = 30
	# any one of the act's four answers to the north
	prerequisite = {
		focus = mltd_wbh_the_tide_wall
		focus = mltd_brk_the_drowned_covenant
		focus = mltd_bdt_hail_the_drowned_king
		focus = mltd_bdt_drown_the_dance
	}
	available = {
		# 50 or more, or from 2281 so that an MLT stalled in the north is not stranded short of the rest of the story (round-27
		# review: run 8's AI held 35-39 states for the two years after its Final Ritual); both are repeated in
		# mltd_tide_turns_south_states_tt
		custom_trigger_tooltip = {
			tooltip = mltd_tide_turns_south_states_tt
			OR = {
				num_of_controlled_states > 49
				date > 2281.1.1
			}
		}
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = 15
	y = 26
	ai_will_do = { factor = 10 }
	completion_reward = {
		add_ideas = mltd_the_northern_waters
		country_event = { id = mltd.30 days = 1 }
		hidden_effect = {
			every_other_country = {
				news_event = { id = mltd.31 days = 2 }
			}
		}
	}
}
