# Rising Tide - AI division templates for the Mirelurk Tribe (MLT)
#
# Five roles of our own, each available_for MLT only. How many divisions of each the AI wants is set by role_ratio in the
# ai_strategy files. OWB's generic roles that also reach MLT (creatures / amphibious_beast, infantry, the garrisons) cannot
# be blocked for it without overriding OWB's files; role_ratio decides between them and these.
# Every target fits OWB's designer: 5 columns x 3 rows (01_defines.lua:445-447), one sub-unit group per column -
# amphibious_beast_creature, mltd_deep_ones and mltd_star_spawn are mutant_group, infantry and militia are infantry.
# No support companies: MLT starts with none unlocked, and the scripted Deep Ones / Star Spawn templates carry none.
# Target templates are named apart from the role and sub-unit tokens (no precedent in vanilla or OWB shares one).
# OWB replace_paths common/ai_templates, so this file only loads after OWB (dependencies in descriptor.mod).

# Coral Host: the line division, 4 mirelurks + 5 infantry (20 wide, 4 columns). Late (mltd_ai_late_army: the king or armour
# form, and the NCR beaten or 2285) the role turns into an all-mirelurk division, the Tide Wall's 8, and its divisions are
# upgraded in the field as mirelurk stock allows (can_upgrade_in_field and replace_with, vanilla _documentation.md:81-117):
# run 2 ended with 102 Coral Hosts and 75 Shell Levies beside 138 Tide Walls.
mltd_coral_host_templates = {
	available_for = { MLT }
	role = mltd_coral_host

	upgrade_prio = {
		factor = 60
		modifier = {
			factor = 0
			NOT = { has_tech = amphibious_beast_unlock_tech }
		}
	}

	mltd_coral_host_target = {
		reinforce_prio = 1
		upgrade_prio = {
			factor = 2
			modifier = {
				factor = 0
				mltd_ai_late_army = yes
			}
		}
		can_upgrade_in_field = {
			mltd_ai_late_army = yes
			has_equipment = { amphibious_beast_equipment > 1000 }
		}
		target_template = {
			regiments = {
				amphibious_beast_creature = 4
				infantry = 5
			}
		}
		replace_at_match = 1.5
		replace_with = mltd_coral_host_late
		target_min_match = 0.0
	}

	mltd_coral_host_late = {
		reinforce_prio = 1
		enable = { mltd_ai_late_army = yes }
		upgrade_prio = {
			factor = 1
			modifier = {
				factor = 5
				mltd_ai_late_army = yes
			}
		}
		target_template = {
			regiments = {
				amphibious_beast_creature = 8
			}
		}
	}
}
