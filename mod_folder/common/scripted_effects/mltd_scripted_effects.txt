# The Final Ritual: every gift becomes permanent. remove_mission skips timeout_effect, so the
# running gifts are not switched off before the variables are pinned to their full values.
# mltd_gift_manpower_var (the modifier's weekly_manpower, set by the ritual focuses) is deliberately
# NOT reset here - it must survive the Final Ritual.
mltd_all_gifts_permanent = {
	custom_effect_tooltip = mltd_all_gifts_permanent_tt
	hidden_effect = {
		if = {
			limit = { has_active_mission = mltd_gift_of_the_tide_mission }
			remove_mission = mltd_gift_of_the_tide_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_shell_mission }
			remove_mission = mltd_gift_of_the_shell_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_current_mission }
			remove_mission = mltd_gift_of_the_current_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_brood_mission }
			remove_mission = mltd_gift_of_the_brood_mission
		}
		if = {
			limit = { has_active_mission = mltd_gift_of_the_abyss_mission }
			remove_mission = mltd_gift_of_the_abyss_mission
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = mltd_gifts_from_the_deep } } }
			add_dynamic_modifier = { modifier = mltd_gifts_from_the_deep }
		}
		set_variable = { mltd_gift_tide_var = 0.20 }
		set_variable = { mltd_gift_shell_var = 0.20 }
		set_variable = { mltd_gift_current_var = 0.20 }
		set_variable = { mltd_gift_brood_var = 0.30 }
		set_variable = { mltd_gift_abyss_var = 0.20 }
		set_variable = { mltd_gifts_active = 0 }
		set_country_flag = mltd_gifts_permanent
		force_update_dynamic_modifier = yes
	}
}

# by name there - rename "Deep Ones" / "Star Spawn" in both places together.
# Before the template is created the hidden equipment tech (mltd_technologies.txt) is granted:
# create_unit ... start_equipment_factor only issues equipment the country has UNLOCKED, and the
# reward tech that also unlocks it only comes at event mltd.8 (fired by mltd_the_deep_ones_walk) / mltd_the_final_ritual - without this the summoned divisions

		limit = { NOT = { has_template_containing_unit = mltd_deep_ones } }
		division_template = {
			name = "Deep Ones"
			is_locked = yes
			priority = 2
			template_counter = 256

# the division's need (8 x 20 x 0.2 = 32 Deep Ones, 8 x 10 x 0.2 = 16 Star Spawn), a small reinforcement pool. Until

		add_equipment_to_stockpile = {

# National population (country scope), recomputed every day by on_daily_MLT: the population of
# every owned state. The ritual focuses gate on it with check_variable and print it with
# [?mltd_national_population|0]; the decisions' gates and AI guards read it too. PREV inside the
# state loop is the country (OWB exodus_effects.txt:392-396 sums state_population_k into
# PREV.core_population_k the same way).
#
# Summed in THOUSANDS (state_population_k, fractional, so exact to the person), then copied back
# into people, capped at 2,147,000. A raw sum of state_population wraps negative past the variable
# ceiling of 2,147,483.647 (Gotchas), which the later story acts reach (the plan puts MLT at 3-4M):
# the Final Ritual's gate, and every act after it, would then fail for good, the Leviathan summon
# would grey out and the AI guards (< 72,000, < 300,000) would read the wrapped value as small.
# Every threshold that reads these two variables is below the cap, so capping changes no test; the
# ritual tooltips' "currently" simply stops at 2,147,000. The _k variables stay for anything that
# needs the true figure.
mltd_update_national_population = {
	set_variable = { mltd_national_population_k = 0 }
	set_variable = { mltd_controlled_population_k = 0 }
	every_owned_state = {
		add_to_variable = { PREV.mltd_national_population_k = state_population_k }
		# The summons gate on this one, because it is exactly the pool
		# mltd_take_population_everywhere spends from - same is_controlled_by / > 500 filter. Gating
		# on the owned total instead would let an occupied MLT clear the gate and then spread the
		# whole toll over the few states it still holds, or, with nothing controlled, pay nothing at
		# all and get the division free.
		if = {
			limit = {
				is_controlled_by = ROOT
				state_population > 500
			}
			add_to_variable = { PREV.mltd_controlled_population_k = state_population_k }
		}
	}
	set_variable = { mltd_national_population = mltd_national_population_k }
	clamp_variable = { var = mltd_national_population max = 2147 }
	multiply_variable = { mltd_national_population = 1000 }
	set_variable = { mltd_controlled_population = mltd_controlled_population_k }
	clamp_variable = { var = mltd_controlled_population max = 2147 }
	multiply_variable = { mltd_controlled_population = 1000 }
}

# The summon toll. Country scope; set mltd_toll_target_k first, in THOUSANDS of people. Takes that
# many IN TOTAL, split across every owned+controlled state over 500 population in proportion to what
# each holds, so the big states carry most of it and none is emptied - a state with a tenth of the
# nation pays a tenth. Two passes: sum the pool, then charge each state population * target / pool.
# Both variables are country variables read from state scope through PREV, the same way the pool is
# accumulated (as in mltd_update_national_population), so nothing depends on a temp variable
# surviving a scope change. add_manpower in state scope is population, and it is used directly
# rather than through OWB's add_state_population so the state tooltips collapse into the one custom
# line the decision shows.
#
# ORDER AND UNITS MATTER. Script variables are 32-bit fixed point with 3 decimals, so they overflow
# above 2,147,483.647 and WRAP NEGATIVE (OWB names the number at TON_farming_scripted_effects.txt:11
# and detects the wrap at coring_button_scripted_triggers.txt:141). Multiplying population by the
# target first would reach 28,000 x 16,000 = 448,000,000, two hundred times over the ceiling, on
# every single state. Dividing first keeps the intermediate at population/pool_k, which can never
# exceed 1,000, and the pool is summed with state_population_k (vanilla's own overflow-avoiding
# variant, dynamic_variables_documentation.md:1168) so the divisor is in thousands. Peak
# intermediate for MLT's kingdom is 372, about 360x under the ceiling, and the charges still sum to
# the target within one person.
mltd_take_population_everywhere = {
	set_variable = { mltd_toll_pool_k = 0 }
	every_owned_state = {
		limit = {
			is_controlled_by = ROOT
			state_population > 500
		}
		add_to_variable = { PREV.mltd_toll_pool_k = state_population_k }
	}
	if = {
		limit = { check_variable = { mltd_toll_pool_k > 0 } }
		every_owned_state = {
			limit = {
				is_controlled_by = ROOT
				state_population > 500
			}
			set_temp_variable = { mltd_toll_temp = state_population }
			divide_temp_variable = { mltd_toll_temp = PREV.mltd_toll_pool_k }
			multiply_temp_variable = { mltd_toll_temp = PREV.mltd_toll_target_k }
			multiply_temp_variable = { mltd_toll_temp = -1 }
			round_temp_variable = mltd_toll_temp
			# The gate is a variable refreshed once a day, so between refreshes the pool can shrink
			# (states lost, occupied, or drained by the ritual toll). Floor every charge at whatever
			# leaves the state 500 people, so no single state can ever be emptied by a summoning.
			set_temp_variable = { mltd_toll_floor_temp = state_population }
			add_to_temp_variable = { mltd_toll_floor_temp = -500 }
			multiply_temp_variable = { mltd_toll_floor_temp = -1 }
			clamp_temp_variable = { var = mltd_toll_temp min = mltd_toll_floor_temp }
			add_manpower = mltd_toll_temp
		}
	}
}

# The ritual toll (on_daily_MLT while mltd_the_grand_ritual / mltd_the_final_ritual is held).
# Country scope. The factor is copied to a temp variable before the state loop (temp variables
# survive scope changes - OWB exodus_effects.txt sets pop_add then reads it in capital_scope).
# Per state: population * factor, negated, at least 5 (max = -5), rounded, then state-scope
# add_manpower, which is population (tlaloc_on_actions.txt does the same with state_pop).
mltd_ritual_daily_toll = {
	set_temp_variable = { mltd_toll_factor_temp = mltd_ritual_toll_factor }
	every_owned_state = {
		limit = { state_population > 500 }
		set_temp_variable = { mltd_toll_temp = state_population }
		multiply_temp_variable = { mltd_toll_temp = mltd_toll_factor_temp }
		multiply_temp_variable = { mltd_toll_temp = -1 }
		clamp_temp_variable = { var = mltd_toll_temp max = -1 }
		round_temp_variable = mltd_toll_temp
		add_manpower = mltd_toll_temp
	}
}

# ---------------------------------------------------------------- Cult Infiltration
# All four run in the scope of the country that holds (or is about to hold) the cult. MLT is named
# by tag, never ROOT: mltd_cult_refresh also runs from on_war_relation_added, where ROOT can be
# either side.
# Each is defined before anything that calls it - a console reload logged the forward references
# (mltd_cult_add_strength -> mltd_cult_refresh, mltd_cult_weekly -> mltd_cult_dissolve) as unknown.
# Variables on that country:
#   mltd_cult_strength          0-100, shown as a percentage
#   mltd_cult_decay             points it loses at the next weekly tick
#   mltd_cult_threat_big/_small/_air  the war penalties in percentage points, for loc (0.25 / 0.15 / 0.5 per point)
#   mltd_cult_*_var             the five mltd_cult_infiltration inputs, 0 unless at war with MLT
# Country flag mltd_cult_sheltered (set on its targets by each story act's spying head,
# mltd_act*_focus.txt) halves the decay.
# MLT keeps mltd_cult_total_strength, the sum of every cult's strength, which the Call decisions pay
# out on.

# The last of the faithful are gone: the country is clean, and a new cult can be founded there.
mltd_cult_dissolve = {
	remove_dynamic_modifier = { modifier = mltd_cult_infiltration }
	clear_variable = mltd_cult_strength
	clear_variable = mltd_cult_decay
	clear_variable = mltd_cult_threat_big
	clear_variable = mltd_cult_threat_small
	clear_variable = mltd_cult_threat_air
	clear_variable = mltd_cult_stability_var
	clear_variable = mltd_cult_war_support_var
	clear_variable = mltd_cult_org_var
	clear_variable = mltd_cult_factory_var
	clear_variable = mltd_cult_air_var
}

# Recomputes the weekly decay and the war penalties from the current strength s.
# decay = 0.1 + s * s / 3000 points a week: 0.13 at 10 %, 0.63 at 40 %, 0.93 at 50 %, 2.8 at 90 %, 3.43 at 100 %, so a
# young cult barely fades and holding 90 costs ~3x what holding 50 does. Halved in a country a story
# act's head has sheltered (mltd_cult_sheltered); tripled while at war with MLT: the faithful rise,
# and rising exposes them. Until 2026-09-23 it was 0.5 + s * s / 2500 (0.54 at 10 %, 1.5 at 50 %),
# under which a cult nurtured back to back by one operative barely grew: its floor alone took over a
# third of a nurture's +12 across the operation's 60 days.
# s * s peaks at 10,000, far under the 2,147,483 variable ceiling.
mltd_cult_refresh = {
	set_variable = { mltd_cult_decay = mltd_cult_strength }
	multiply_variable = { mltd_cult_decay = mltd_cult_strength }
	divide_variable = { mltd_cult_decay = 3000 }
	add_to_variable = { mltd_cult_decay = 0.1 }
	if = {
		limit = { has_country_flag = mltd_cult_sheltered }
		multiply_variable = { mltd_cult_decay = 0.5 }
	}
	if = {
		limit = { has_war_with = MLT }
		multiply_variable = { mltd_cult_decay = 3 }
	}
	set_variable = { mltd_cult_threat_big = mltd_cult_strength }

# mltd_cult_rebuild_list: the cults abroad, strongest first, for the Cult Infiltration category's
# description (GetMltdCultList, common/scripted_localisation). MLT scope. Rebuilt rather than kept
# up piecemeal: every_country skips countries that no longer exist, so annexed ones drop out. Called
# daily by on_daily_MLT once The Spreading Cult is done, weekly after the decay loop, by both cult
# operations and by the focus mltd_the_spreading_cult. It also sums mltd_cult_total_strength, which
# the Call decisions pay out on.
# The loc reads fixed slots on MLT - mltd_cult_row_<k> (the country) and mltd_cult_row_<k>_str (its strength), k = 1-8,
# plus mltd_cult_count and mltd_cult_more - never an array element: stepping into var:array^i from a
# decision category's description found nothing in game. Slots past mltd_cult_count keep stale
# values that the loc never shows. Selection sort with find_highest_in_array and remove_from_array
# by index (OWB caps_scripted_ai.txt:161-218).
mltd_cult_rebuild_list = {
	clear_array = mltd_cult_list
	clear_array = mltd_cult_sort_country
	clear_array = mltd_cult_sort_strength
	set_variable = { mltd_cult_count = 0 }
	set_variable = { mltd_cult_total_strength = 0 }
	every_country = {
		limit = { has_variable = mltd_cult_strength }
		add_to_array = { ROOT.mltd_cult_sort_country = THIS }
		add_to_array = { ROOT.mltd_cult_sort_strength = mltd_cult_strength }
		add_to_variable = { ROOT.mltd_cult_total_strength = mltd_cult_strength }
	}
	while_loop_effect = {
		limit = { check_variable = { mltd_cult_sort_country^num > 0 } }
		find_highest_in_array = { array = mltd_cult_sort_strength index = mltd_cult_hi }
		add_to_variable = { mltd_cult_count = 1 }
		var:mltd_cult_sort_country^mltd_cult_hi = {
			if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 1 } }
				set_variable = { ROOT.mltd_cult_row_1 = THIS }
				set_variable = { ROOT.mltd_cult_row_1_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 2 } }
				set_variable = { ROOT.mltd_cult_row_2 = THIS }
				set_variable = { ROOT.mltd_cult_row_2_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 3 } }
				set_variable = { ROOT.mltd_cult_row_3 = THIS }
				set_variable = { ROOT.mltd_cult_row_3_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 4 } }
				set_variable = { ROOT.mltd_cult_row_4 = THIS }
				set_variable = { ROOT.mltd_cult_row_4_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 5 } }
				set_variable = { ROOT.mltd_cult_row_5 = THIS }
				set_variable = { ROOT.mltd_cult_row_5_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 6 } }
				set_variable = { ROOT.mltd_cult_row_6 = THIS }
				set_variable = { ROOT.mltd_cult_row_6_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 7 } }
				set_variable = { ROOT.mltd_cult_row_7 = THIS }
				set_variable = { ROOT.mltd_cult_row_7_str = mltd_cult_strength }
			}
			else_if = {
				limit = { check_variable = { ROOT.mltd_cult_count = 8 } }
				set_variable = { ROOT.mltd_cult_row_8 = THIS }
				set_variable = { ROOT.mltd_cult_row_8_str = mltd_cult_strength }
			}
		}
		remove_from_array = { array = mltd_cult_sort_country index = mltd_cult_hi }
		remove_from_array = { array = mltd_cult_sort_strength index = mltd_cult_hi }
	}
	set_variable = { mltd_cult_more = mltd_cult_count }
	subtract_from_variable = { mltd_cult_more = 8 }
}

			every_country = {
				limit = {
					has_country_flag = mltd_wbh_faction_moving
					is_subject = yes
				}
				mltd_wbh_bring_into_the_covenant = yes
			}
			every_country = {
				limit = { has_country_flag = mltd_wbh_faction_moving }
				clr_country_flag = mltd_wbh_faction_moving
			}
		}
	}
