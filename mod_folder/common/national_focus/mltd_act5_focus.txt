# is_ai modifiers that waited on the NCR's stage are gone, because the tree now enforces the order.

# their AI readiness hold (mltd_ai_army_ready; since the round-27 review in `available`,

# Round 28b (the play-test fixes):
# - The four wars' thefts are a quarter of what each victim holds in store, not a flat 600-800: a
#   flat lot was ~5 % of a year's arms output when the plan opens this act (~60 military factories,
#   CONQUEST_PLAN.txt's 2282-05 snapshot) and MLT's line is turning into mirelurks by then, while a
#   quarter of the enemy's reserve on the war's first day weighs the same in any year. Each share is
#   read into a temp variable for the preview (the tooltip prints it) and again at completion; the
#   transfers are hidden. The quarter is a literal in each focus and in its tooltip - change both.
# - Round 28c: the thefts pay in what the tribe fields from this act on. The victim still loses its
#   stores, with no producer on the removal, but the plan's levies become all-mirelurk Tide Walls at
#   this act's chapter, so rifles are no use to us: the haul is melted down into harness and
#   shell-plate for the brood - one mirelurk (the archetype amphibious_beast_equipment, our best
#   unlocked breed) for every five weapons or crates taken, and one for every suit of power armour,
#   about their worth in IC. A quarter of a Legion depot of ~6,000 rifles and ~1,200 crates is ~360
#   mirelurks, ~6,500 IC: about a tenth of a year's output at the plan's ~60 military factories,
#   taken from the enemy on the war's first day. The tooltips print the mirelurks as the stores
#   stand.
# - Round 28c (the review): a theft's old fallback is now its floor, paid whenever the haul comes to
#   less than the fallback is worth, so a victim with a small store never pays less than one with
#   none (the victim still loses what was taken).
#   100 caps buy 125 mirelurks at the Wet Market's dearest lot (200 caps for 250), so The Sea Against Mars and Blood in the
#   Water pay their 100 caps under 125 mirelurks; 25 political power is 20 caps at OWB's own rate
#   (add_caps pays 1.25 political power a cap with the caps rule off), about 25 mirelurks, so the
#   Rites and the Crusade pay their +25 under 25.
# - Every effect that hangs on a condition that can still change while the focus runs (Caesar's health, the Cult of Mars,
#   Lost Hills' quarrel with the NCR, the White Legs' gods, the road war, Heaven's holy wars) is
#   previewed as a condition, with its native lines (effect_tooltip), and applied in hidden_effect,
#   so the tooltip is true whichever way it falls.
#   A condition that can no longer change - a focus the other country has completed, a country
#   gone - still shows as it is.
#
# Round 28c (readability): these are the longest rewards in the mod - a war focus can render sixteen or seventeen lines - so
# each is broken into blocks with the house spacer, custom_effect_tooltip = mltd_newline_tt (loc " \n"; CLAUDE.md > The
# rituals, and the invitations in mltd_act3_focus.txt). The order of the blocks is what we gain,
# what the target loses, the intelligence, the war, then the event - never a reordering of the
# effects themselves, only blank lines between them. The four heads share one shape: the foothold
# and the cult it shelters, then what that branch's merged focus brings, then the intelligence
# grants. The four wars: the declaration, then the theft, then each outcome of a condition as a
# block of its own - a preview that lists three ways Caesar's health or Lost Hills' quarrel may fall
# needs the blank lines to be read at all. A spacer sits inside the branch it separates wherever the
# branch can fail, so it never prints for a branch that did not fire, and never last in a reward or
# a branch. Presentation only - no number, gate or effect changed. Left alone, at two lines: the
# chapter, Mars in the Water.
#
# 2026-09-24 (the user: war focuses should bring bonuses against the country and an immediate buff,
# not stability and war-support losses of little consequence; and too much text): each of the four
# wars declares, then - against a target that is not MLT's subject - grants its 180-day idea, which

# The cult spies on Flagstaff and turns a frumentarius (merges The Drowned Frumentarius)
shared_focus = {
	id = mltd_ces_what_the_river_carries
	icon = GFX_goal_EXE_something_fishy
	prerequisite = { focus = mltd_mars_in_the_water }
	x = -3
	y = 1
	relative_position_id = mltd_mars_in_the_water
	cost = 30
	ai_will_do = { factor = 3 }
	# 2026-09-23: bypassed once CES is gone - with its nation gone it paid only its fallback
	bypass = {
		NOT = { country_exists = CES }
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	completion_reward = {
		mltd_intel_foothold = yes
		# round 17: the cults of M'lyeh in this branch's countries wither half as fast
		# (mltd_cult_refresh)
		if = {
			limit = { country_exists = CES }
			custom_effect_tooltip = mltd_ces_cult_sheltered_tt
			hidden_effect = {
				CES = { set_country_flag = mltd_cult_sheltered }
			}
		}
		# round 28c: the foothold and the sheltered cult above, then what the frumentarius brings,
		# then what the Legion loses.
		# The if/else below always renders one branch, so this spacer can stand outside it
		custom_effect_tooltip = mltd_newline_tt
		if = {
			limit = { has_dlc = "La Resistance" }
			# the drowned frumentarius (merged): an operative once the agency exists, as before.
			# Round 28c review: the branch is chosen when the focus completes, and an agency can
			# still be founded from the intelligence screen while it runs, so with none yet the
			# preview writes both outcomes (mltd_ces_frumentarius_no_agency_tt, the shape of
			# mltd_intel_foothold's no-agency branch) over a hidden +50 PP; the 50 is repeated in
			# that key
			if = {
				limit = { has_intelligence_agency = yes }
				create_operative_leader = {
					name = mltd_ces_drowned_frumentarius_operative
					gfx = GFX_Portrait_CES_Operator_3
					skill = 2
					traits = { operative_frumentarius }
					bypass_recruitment = no
					available_to_spy_master = yes
					nationalities = { MLT CES }
				}
			}
			else = {
				custom_effect_tooltip = mltd_ces_frumentarius_no_agency_tt
				hidden_effect = { add_political_power = 50 }
			}
			# one intelligence grant (the river's civ 10 / army 10 and the frumentarius' civ 15, at
			# the larger values) and the frumentarius' army token; the Legion gone, the river's +25
			# stands in for both
			if = {
				limit = { country_exists = CES }
				add_operation_token = {
					tag = CES
					token = token_army
				}
				add_intel = {
					target = CES
					civilian_intel = 15
					army_intel = 10
				}
			}
			else = { add_political_power = 25 }
		}
		else = {
			# 25 (What the River Carries) + 50 (The Drowned Frumentarius), their two no-DLC
			# fallbacks
			add_political_power = 75
		}
		if = {
			limit = { country_exists = CES }
			# round 28c: what the Legion loses is its own block, and the spacer is inside the
			# branch, so nothing is separated off when there is no Legion left
			custom_effect_tooltip = mltd_newline_tt
			CES = { add_stability = -0.06 }
		}
	}
}

# The war on the Legion (merges A Crate for Every Cross)
shared_focus = {
	id = mltd_ces_the_sea_against_mars
	icon = GFX_goal_CES_Mars_God
	prerequisite = { focus = mltd_ces_what_the_river_carries }
	x = -3
	y = 2
	relative_position_id = mltd_mars_in_the_water
	cost = 30
	will_lead_to_war_with = CES
	ai_will_do = { factor = 3 }
	available = {
		# round 22: the focus is the declaration, so an AI takes it only when the army is ready -
		# only while it would start a war. Round 27: its stage term (mltd_ai_ncr_beaten) is gone -
		# the chapter already needs Act IV's close, and the term could only hold the AI back once a
		# beaten NCR outlived the peace (has_capitulated clears then), stranding Act V, Act VI and
		# the finale. Since the round-27 review the hold sits here, hidden and AI-only, because an
		# AI takes any listed focus that is available whatever its ai_will_do reads; it also waits
		# while MLT is losing a war (mltd_ai_triggers.txt, section W)
		mltd_ai_may_start_ces_war = yes
	}
	# 2026-09-23: bypassed once CES is gone - with its nation gone it paid only its fallback
	bypass = {
		NOT = { country_exists = CES }
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	completion_reward = {
		if = {
			limit = {
				country_exists = CES
				NOT = { has_war_with = CES }
				CES = { NOT = { is_subject_of = ROOT } }
			}
			declare_war_on = {
				type = annex_everything
				target = CES
			}
		}
		# 2026-09-24 (the user: war focuses should bring bonuses against the country and an
		# immediate buff, not stability and war-support losses of little consequence; and less
		# text): the war, then - against a target that is not MLT's subject - the 180-day idea with
		# +10 % attack and defence against it, The First Wave (60 days) and the theft, one line;
		# with the target MLT's subject, +50 political power. The side branches keyed to the
		# target's own story went with the losses (Caesar's health, his succession war and the Cult
		# of Mars, round 28b-2026-09-23). A Crate for Every Cross (merged): a quarter of each store
		# of at least 100, paid as one mirelurk (the archetype amphibious_beast_equipment, our best
		# unlocked breed) for every five pieces; the quarter and the five are literals here and in
		# mltd_ces_crates_theft_tt.
		# Its caps floor went with the rest
		if = {
			limit = {
				country_exists = CES
				CES = { NOT = { is_subject_of = ROOT } }
			}
			add_timed_idea = {
				idea = mltd_ces_the_drowned_bull
				days = 180
			}
			add_timed_idea = {
				idea = mltd_the_first_wave
				days = 60
			}
			set_temp_variable = { mltd_theft_infantry = 0 }
			if = {
				limit = {
					country_exists = CES
					CES = { has_equipment = { infantry_equipment > 99 } }
				}
				set_temp_variable = { mltd_theft_infantry = CES.num_equipment@infantry_equipment }
				multiply_temp_variable = { mltd_theft_infantry = 0.25 }
				round_temp_variable = mltd_theft_infantry
			}
			set_temp_variable = { mltd_theft_support = 0 }
			if = {
				limit = {
					country_exists = CES
					CES = { has_equipment = { support_equipment > 99 } }
				}
				set_temp_variable = { mltd_theft_support = CES.num_equipment@support_equipment }
				multiply_temp_variable = { mltd_theft_support = 0.25 }
				round_temp_variable = mltd_theft_support
			}
			set_temp_variable = { mltd_theft_brood = mltd_theft_infantry }
			add_to_temp_variable = { mltd_theft_brood = mltd_theft_support }
			divide_temp_variable = { mltd_theft_brood = 5 }
			round_temp_variable = mltd_theft_brood
			custom_effect_tooltip = mltd_ces_crates_theft_tt
			hidden_effect = {
				if = {
					limit = {
						CES = { has_equipment = { infantry_equipment > 99 } }
					}
					set_temp_variable = { mltd_theft_negative = mltd_theft_infantry }
					multiply_temp_variable = { mltd_theft_negative = -1 }
					CES = {
						add_equipment_to_stockpile = {
							type = infantry_equipment
							amount = mltd_theft_negative
						}
					}
				}
				if = {
					limit = {
						CES = { has_equipment = { support_equipment > 99 } }
					}
					set_temp_variable = { mltd_theft_negative = mltd_theft_support }
					multiply_temp_variable = { mltd_theft_negative = -1 }
					CES = {
						add_equipment_to_stockpile = {
							type = support_equipment
							amount = mltd_theft_negative
						}
					}
				}
				if = {
					limit = { check_variable = { mltd_theft_brood > 0 } }
					add_equipment_to_stockpile = {
						type = amphibious_beast_equipment
						amount = mltd_theft_brood
					}
				}
			}
		}
		else = { add_political_power = 50 }
	}
}
