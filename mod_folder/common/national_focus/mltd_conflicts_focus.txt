# Rising Tide (MLT) - conflict sub-trees.
# Shared focuses, pulled into mlt_nf by one `shared_focus = <root>` line each in
# "Mirelurk Tribe (MLT) Focus.txt". Each root opens after mlt_kingdom_of_mlyeh via `available`,
# has absolute x/y and no prerequisite; every other focus hangs off its own root, so the engine
# pulls it in through that prerequisite (a shared focus whose only prerequisite is a national
# focus would not appear). Layout: CLAUDE.md > Conflict sub-trees.

shared_focus = {
	id = mltd_ncr_a_mole_in_shady_sands
	icon = GFX_goal_NCR_Shady_Sands
	cost = 30
	available = {
		has_completed_focus = mlt_kingdom_of_mlyeh
		country_exists = NCR
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	x = 2
	y = 13
	ai_will_do = {
		factor = 3

			NCR = { set_country_flag = mltd_cult_sheltered }

			OR = {
				NOT = { mltd_ai_stage_ncr = yes }
				NOT = { mltd_ai_army_ready = yes }
			}
		}
	}
	will_lead_to_war_with = NCR
	completion_reward = {
		# round 22: the sub-tree ends in the war itself - the focus declares on each living target that is not already at war

		# goal the AI could sit on for years)
		if = {
			limit = {
				country_exists = NCR
				NOT = { has_war_with = NCR }
				NCR = { NOT = { is_subject_of = ROOT } }

			declare_war_on = {
				type = annex_everything
				target = NCR
			}
		}
		if = {
			limit = {
				has_dlc = "La Resistance"
				country_exists = NCR
			}
			steal_random_tech_bonus = {
				folder = fallout_industry_folder
				ahead_reduction = 1
				bonus = 1.0
				base_bonus = 0.5

				name = mltd_ncr_the_turbines_sing
				bonus = 0.5
				uses = 1
				category = industry
			}
		}
		if = {
			limit = { has_global_flag = fbhd_legion_overall_vic }
			custom_effect_tooltip = mltd_ncr_turbines_rout_tt
			if = {
				limit = {
					country_exists = NCR

				OR = {

				}
