country_event = {
	immediate = { log = "[GetDateText]: [Root.GetName]: event nf_mlt.4" }
	id = nf_mlt.4
	title = nf_mlt.4.t
	desc = nf_mlt.4.d

	is_triggered_only = yes

	option = {
		name = nf_mlt.4.a
		ai_chance = {
			factor = 20 #This is a good deal - unless...
			modifier = {
				#Lil' harder to make deals with mutants
				original_tag = TRL
				factor = 0.75
			}
			modifier = {
				#Don't do it if we are about to attack MLT, or are fighting a defensive war
				OR = {
					has_wargoal_against = MLT
					has_war_with = MLT
					has_war_with = WBH
					has_defensive_war = yes
				}
				factor = 0
			}
			modifier = {
				#These states are important for supply reasons
				owns_state = 37
				owns_state = 506
				factor = 0
			}
			modifier = {
				#Should we even be listening to these guys?
				ic_ratio = {
					tag = MLT
					ratio > 4
				}
				factor = 0
			}
			modifier = {
				#Do we even want what they are offering?
				check_variable = {
					var = THIS.resource@water
					value = 32
					compare = greater_than
				}
				factor = 0
			}
		}
		if = {
			limit = {
				owns_state = 186
			}
			MLT = {
				capital_scope = {
					add_resource = {
						type = water
						amount = -8
					}
				}
			}
			capital_scope = {
				add_resource = {
					type = water
					amount = 8
				}
			}
			186 = {
				remove_core_of = ROOT
				remove_claim_by = ROOT
			}
			MLT = {
				transfer_state = 186
				186 = {
					if = {
						limit = {
							is_core_of = RBT
						}
						remove_core_of = RBT
					}
				}
			}
		}
		if = {
			limit = {
				owns_state = 450
			}
			MLT = {
				capital_scope = {
					add_resource = {
						type = water
						amount = -8
					}
				}
			}
			capital_scope = {
				add_resource = {
					type = water
					amount = 8
				}
			}
			450 = {
				remove_core_of = ROOT
				remove_claim_by = ROOT
			}
			MLT = {
				transfer_state = 450
				450 = {
					if = {
						limit = {
							is_core_of = RBT
						}
						remove_core_of = RBT
					}
				}
			}
		}
		if = {
			limit = {
				owns_state = 493
			}
			MLT = {
				capital_scope = {
					add_resource = {
						type = water
						amount = -8
					}
				}
			}
			capital_scope = {
				add_resource = {
					type = water
					amount = 8
				}
			}
			493 = {
				remove_core_of = ROOT

			}

				transfer_state = 493
				493 = {
					if = {
						limit = {
							is_core_of = RBT
						}
						remove_core_of = RBT
					}
