			factor = 1

	# Offerings to the Deep: population traded to the sea for what the drowned left behind.
	# Population is taken with OWB's add_state_population (temp var pop_add, 00_scripted_effects.txt:509)
	# from any owned+controlled state large enough, exactly like the summons; caps go through OWB's
	# add_caps (caps_scripted_effects.txt:33, temp var caps_to_add - it writes its own tooltip, plays the
	# sound, clamps the balance and refreshes the top bar, and pays 1.25x political power instead when the
	# caps game rule is off). Both are on a visible timed-flag cooldown rather than days_re_enable.
	mltd_offering_of_the_drowned = {
		icon = GFX_decision_skull
		allowed = {
			original_tag = MLT
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = mltd_offering_state_pop_3000_tt
				any_owned_state = {
					is_controlled_by = ROOT
					state_population > 3000
				}
			}
			custom_trigger_tooltip = {
				tooltip = mltd_offering_drowned_cooldown_tt
				NOT = { has_country_flag = mltd_offering_drowned_cooldown }
			}
		}
		cost = 25
		fire_only_once = no
		ai_will_do = {
			factor = 2
			modifier = {
				factor = 0
				NOT = { any_owned_state = { is_controlled_by = ROOT state_population > 6000 } }
			}
		}
		complete_effect = {
			random_owned_controlled_state = {
				limit = { state_population > 3000 }
				set_temp_variable = { pop_add = -1000 }
				add_state_population = yes
			}
			set_temp_variable = { caps_to_add = 100 }
			add_caps = yes
			custom_effect_tooltip = mltd_offering_drowned_cooldown_after_tt
			hidden_effect = {
				set_country_flag = { flag = mltd_offering_drowned_cooldown value = 1 days = 60 }
			}
		}
	}
