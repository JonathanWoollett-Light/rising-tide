# The Deep Ones Walk: the Deep Ones become trainable, the first Star Spawn division rises and its summon decision opens (fired by focus mltd_the_deep_ones_walk)
country_event = {
	immediate = { log = "[GetDateText]: [Root.GetName]: event mltd.8" }
	id = mltd.8
	title = mltd.8.t
	desc = mltd.8.d
	picture = GFX_event_mirelurk_red_death_shipwrecker

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = mltd.8.a
		# warbike_unlock_tech = The Deep Ones (id reuses OWB's orphan reward-tab gridbox)
		set_technology = { warbike_unlock_tech = 1 }
		add_ideas = mltd_children_of_the_deep
		mltd_ensure_deep_ones_template = yes
		if = {
			limit = { has_template_containing_unit = mltd_deep_ones }
			set_division_template_lock = {
				division_template = "Deep Ones"
				is_locked = no
			}
		}
		custom_effect_tooltip = mltd_deep_ones_template_unlocked_tt
		mltd_spawn_star_spawn_division = yes
		custom_effect_tooltip = mltd_star_spawn_decision_unlocked_tt
		custom_effect_tooltip = mltd_deep_ones_summon_replaced_tt
		set_country_flag = mltd_star_spawn_summon_unlocked
	}
}
