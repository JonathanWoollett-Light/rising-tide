on_actions = {
	# Every day: recompute mltd_national_population (the ritual focuses gate on it), then, while
	# either ritual idea is held, take the toll from every owned state
	# (both effects in common/scripted_effects/mltd_scripted_effects.txt).
	on_daily_MLT = {
		effect = {
			mltd_update_national_population = yes
			if = {
				limit = {
					OR = {
						has_idea = mltd_the_grand_ritual
						has_idea = mltd_the_final_ritual
					}
				}
				mltd_ritual_daily_toll = yes
			}
			# the Cult Infiltration category's list of cults (an annexed country drops out the next day)
			if = {
				limit = { has_completed_focus = mltd_the_spreading_cult }
				mltd_cult_rebuild_list = yes
			}

			# Round 25: the Broken Coast's landings on Haida Gwaii (event mltd.27). BRK's own OWB focus
			# brk_nf_lessons_of_war sends it against the Haida Confederation, whose whole country is one island: nine
			# land provinces, every one coastal, only two with a port (mltd_brk_on_haida_gwaii). BRK's AI lands wherever
			# its planner picks - usually a portless beach - and the landing starves and is thrown back; it lands again,
			# and MLT's volunteers go down with each landing. The engine chooses the landing province, so nothing in
			# script can aim it at a port. Instead, while MLT is engaged with the Broken Coast and both sides are AIs,
			# watch for a foothold - daily, so a landing thrown back within the week is not missed - and make peace the day
			# BRK loses its last one. "Pushed off" means landed, then lost: BRK holds no Haida province before its first
			# landing either, so a plain control test would end the war the day it began.
			# The flag is cleared whenever that war is not running, so a later war starts clean. The watch starts with
			# Salt on the Broken Coast, which since round 27 is an Act II head (mltd_act2_focus.txt), after The Deep Ones Walk.
			if = {
				limit = {
					has_global_flag = mltd_brk_landed_on_haida
					NOT = {
						country_exists = BRK
						country_exists = HAI
						BRK = { has_war_with = HAI }
					}
				}
				clr_global_flag = mltd_brk_landed_on_haida
			}
			if = {
				limit = {
					has_completed_focus = mltd_brk_salt_on_the_broken_coast
					country_exists = BRK
					country_exists = HAI
					BRK = {
						is_ai = yes
						has_war_with = HAI
					}
					HAI = { is_ai = yes }
				}
				if = {
					limit = {
						NOT = { has_global_flag = mltd_brk_landed_on_haida }
						BRK = { mltd_brk_on_haida_gwaii = yes }
					}
					set_global_flag = mltd_brk_landed_on_haida
					if = {
						limit = { mltd_telemetry_on = yes }
						log = "MLTD [?mltd_tm_day|0] HAIDA state=landed"
					}
				}
				else_if = {
					limit = {
						has_global_flag = mltd_brk_landed_on_haida
						BRK = { NOT = { mltd_brk_on_haida_gwaii = yes } }
					}
					clr_global_flag = mltd_brk_landed_on_haida
					if = {
						limit = { mltd_telemetry_on = yes }
						log = "MLTD [?mltd_tm_day|0] HAIDA state=pushed_off"
					}
					HAI = { country_event = { id = mltd.27 } }
				}
			}
		}
	}

	# Every month: roll the Lovecraftian flavour pool (events mltd.9-19). Each event's own trigger
	# calls mltd_flavour_can_fire (common/scripted_triggers/mltd_scripted_triggers.txt), and each sets
	# the shared 120-day cooldown flag mltd_flavour_cooldown when it fires. 50 = 0 is the no-event
	# sink (vanilla 00_on_actions.txt:374; Millennium Dawn 99_SYR_on_actions.txt:4).
	on_monthly_MLT = {
		# The Wet Market restocks while the Mirelurk Tribe exists (the stock is global: every buyer draws on it), topping
		# up only below 2,500 so lots sold into it past that are kept.
		effect = {
			if = {
				limit = {
					has_global_flag = mltd_wet_market_open
					check_variable = { global.mltd_wet_market_stock < 2500 }
				}
				add_to_variable = { global.mltd_wet_market_stock = 250 }
				clamp_variable = { var = global.mltd_wet_market_stock min = 0 max = 2500 }
			}
		}
		random_events = {
			50 = 0
			10 = mltd.9
			10 = mltd.10
			10 = mltd.11
			10 = mltd.12
			10 = mltd.13
			10 = mltd.14
			10 = mltd.15
			10 = mltd.16
			10 = mltd.17
			10 = mltd.18
			10 = mltd.19
		}
	}

	# Every week: every cult of M'lyeh abroad loses its decay (faster the stronger it is, three times as
	# fast while at war with us, half as fast where a story act's focus shelters it) and dissolves below 1 %
	# (mltd_cult_weekly); then the
	# category's list of cults is rebuilt (mltd_cult_rebuild_list). An AI Mirelurk Tribe then surveys the map for its
	# strategy (mltd_ai_survey), re-picks the target of its cult operations and may buy a lot from The Wet Market (the three
	# blocks at the end).
	on_weekly_MLT = {
		effect = {
			every_country = {
				limit = { has_variable = mltd_cult_strength }
				mltd_cult_weekly = yes
			}
			mltd_cult_rebuild_list = yes

			# AI only: borders, the army's size, the north and pockets, read by common/ai_strategy
			# (common/scripted_effects/mltd_ai_survey_effects.txt).
			if = {
				limit = { is_ai = yes }
				mltd_ai_survey = yes
			}

			# AI only: infantry weapons and support equipment from OWB's sellers, the player's own purchase
			# (mltd_ai_buy_line_equipment, common/scripted_effects/mltd_ai_survey_effects.txt)
			if = {
				limit = {
					is_ai = yes
					has_global_flag = caps_enabled_global_flag
					has_variable = caps_number_display
				}
				mltd_ai_buy_line_equipment = yes
			}

			# AI only: the target of the cult operations, read by their strategies' enable (common/ai_strategy). The same
			# weekly pick as OWB's update_operation_ai (scripted_effects/operation_strat_effects.txt:5-24): drop a target that
			# is gone, capitulated or allied, then re-score with mltd_ai_cult_target_scorer (common/scorers/country).
			if = {
				limit = {
					is_ai = yes
					has_dlc = "La Resistance"
					has_intelligence_agency = yes
					has_completed_focus = mltd_the_spreading_cult
				}
				if = {
					limit = {
						NOT = { check_variable = { mltd_ai_cult_target = 0 } }
						var:mltd_ai_cult_target = {
							OR = {
								exists = no
								has_capitulated = yes
								is_ally_with = PREV
							}
						}
					}
					set_variable = { mltd_ai_cult_target = 0 }
				}
				# unverified: the vanilla documentation (effects_documentation.md:4483-4507) says this form stores a temp
				# variable and the _temp form a variable; OWB reads its result from a strategy's enable, so it persists. Also
				# unverified: that the variable is left as it was when no country scores above 0.
				get_highest_scored_country = {
					scorer = mltd_ai_cult_target_scorer
					var = mltd_ai_cult_target
				}
				# the Broken Coast while MLT courts it (round 21): the Drowned Covenant waits on a cult there at 50 %, and the
				# scorer's arrays (neighbours, enemies, majors) seldom hold it. Not over the NCR's cult before the NCR's war.
				if = {
					limit = {
						mltd_ai_courting_brk = yes
						network_national_coverage = { target = BRK value > 0 }
						BRK = {
							OR = {
								NOT = { has_variable = mltd_cult_strength }
								check_variable = { mltd_cult_strength < 60 }
							}
						}
						OR = {
							NOT = { has_completed_focus = mltd_the_deep_ones_walk }
							NOT = { country_exists = NCR }
							has_war_with = NCR
							NCR = {
								has_variable = mltd_cult_strength
								check_variable = { mltd_cult_strength > 39.9 }
							}
						}
					}
					set_variable = { mltd_ai_cult_target = BRK }
				}
				# else the Bone Dancers, once they have crowned the Odious King and The Bone Shore has opened their branch (round-27
				# review): Hail the Drowned King waits on a cult there at 40 %, and the scorer seldom picks them either. Nurtured to
				# 50 %, ten points over the gate as the Broken Coast's 60 is over its 50; the Broken Coast comes first, so the two
				# never overwrite each other week to week. The same NCR-first guard.
				else_if = {
					limit = {
						mltd_ai_courting_bdt = yes
						has_completed_focus = mltd_bdt_the_bone_shore
						NOT = { has_completed_focus = mltd_bdt_hail_the_drowned_king }
						BDT = { has_completed_focus = bdt_hail_to_the_king }
						network_national_coverage = { target = BDT value > 0 }
						BDT = {
							OR = {
								NOT = { has_variable = mltd_cult_strength }
								check_variable = { mltd_cult_strength < 50 }
							}
						}
						OR = {
							NOT = { has_completed_focus = mltd_the_deep_ones_walk }
							NOT = { country_exists = NCR }
							has_war_with = NCR
							NCR = {
								has_variable = mltd_cult_strength
								check_variable = { mltd_cult_strength > 39.9 }
							}
						}
					}
					set_variable = { mltd_ai_cult_target = BDT }
				}
			}

			# AI only, once its army turns to mirelurks (mltd_ai_late_army; until then its caps go on the line's infantry weapons):
			# The Wet Market's GUI is is_ai = no, so an AI Mirelurk Tribe buys here, exactly as the buy button does
			# (common/scripted_guis/mltd_wet_market_gui.txt, whose prices these repeat). One lot of the best variant it has
			# the tech for, while its stockpile holds under a lot, with over 300 caps in hand and 150 left after paying. OWB's
			# own AI floor (organization_scripted_effects.txt:654-655) also asks for positive caps income, which a large army's
			# upkeep turns negative (caps_scripted_effects.txt:425-537) and the offerings' caps never count towards; the 150
			# left keeps the balance clear of OWB's bankruptcy test. The shared 40-day flag paces it.
			if = {
				limit = {
					is_ai = yes
					mltd_ai_late_army = yes
					has_global_flag = mltd_wet_market_open
					NOT = { has_country_flag = caps_market_prevented }
					NOT = { has_country_flag = mltd_wet_market_recent_trade }
					check_variable = { caps_number_display > 300 }
					has_equipment = { amphibious_beast_equipment < 250 }
				}
				# the armour and king forms exclude each other (OWB tech_creatures.txt:840-842, :910-912)
				if = {
					limit = { has_tech = amphibious_beast_king_form }
					set_temp_variable = { mltd_wm_price = 200 }
					set_temp_variable = { mltd_wm_ai_variant = 4 }
				}
				else_if = {
					limit = { has_tech = amphibious_beast_armor_form }
					set_temp_variable = { mltd_wm_price = 200 }
					set_temp_variable = { mltd_wm_ai_variant = 3 }
				}
				else_if = {
					limit = { has_tech = amphibious_beast_intermediate_form }
					set_temp_variable = { mltd_wm_price = 180 }
					set_temp_variable = { mltd_wm_ai_variant = 2 }
				}
				else = {
					set_temp_variable = { mltd_wm_price = 150 }
					set_temp_variable = { mltd_wm_ai_variant = 1 }
				}
				set_temp_variable = { mltd_wm_ai_caps_left = caps_number_display }
				subtract_from_temp_variable = { mltd_wm_ai_caps_left = mltd_wm_price }
				if = {
					limit = {
						check_variable = { var = mltd_wm_ai_caps_left value = 150 compare = greater_than_or_equals }
						mltd_wet_market_can_buy = yes
					}
					mltd_wet_market_pay = yes
					if = {
						limit = { check_variable = { mltd_wm_ai_variant = 4 } }
						add_equipment_to_stockpile = {
							type = amphibious_beast_equipment_king
							amount = 250
							producer = MLT
						}
					}
					else_if = {
						limit = { check_variable = { mltd_wm_ai_variant = 3 } }
						add_equipment_to_stockpile = {
							type = amphibious_beast_equipment_armor
							amount = 250
							producer = MLT
						}
					}
					else_if = {
						limit = { check_variable = { mltd_wm_ai_variant = 2 } }
						add_equipment_to_stockpile = {
							type = amphibious_beast_equipment_2
							amount = 250
							producer = MLT
						}
					}
					else = {
						add_equipment_to_stockpile = {
							type = amphibious_beast_equipment_1
							amount = 250
							producer = MLT
						}
					}
				}
			}
		}
	}

	# A war between MLT and a country with a cult switches the cult's penalties on at once instead of at
	# the next weekly tick. ROOT is the attacker, FROM the defender (vanilla 00_on_actions.txt:4911-4913).
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					tag = MLT
					FROM = { has_variable = mltd_cult_strength }
				}
				FROM = { mltd_cult_refresh = yes }
			}
			if = {
				limit = {
					has_variable = mltd_cult_strength
					FROM = { tag = MLT }
				}
				mltd_cult_refresh = yes
			}
		}
	}

	# Round 27, the story acts: a capitulation to MLT is recorded on MLT, so that an act's close outlives the peace -
	# has_capitulated clears once the country leaves its last war, and a rump that survives the peace would otherwise never
	# count as beaten. The NCR (mltd_ncr_capitulated) opens Act IV's close, mltd_when_shady_sands_falls (mltd_act4_focus.txt);
	# the Legion (mltd_ces_capitulated) Act V's, mltd_when_the_legion_breaks (mltd_act5_focus.txt); the two Texan tags
	# (mltd_tbh_capitulated, mltd_lns_capitulated) Act VI's, through mltd_texas_beaten. ROOT is the capitulated country, FROM
	# the winner (vanilla 00_on_actions.txt:931); vanilla tests has_war_with against the capitulated country inside this
	# on_action (:937-938), and FROM = MLT covers MLT's own win if it does not hold.
	on_capitulation = {
		effect = {
			if = {
				limit = {
					OR = {
						has_war_with = MLT
						FROM = { tag = MLT }
					}
				}
				if = {
					limit = { tag = NCR }
					MLT = { set_country_flag = mltd_ncr_capitulated }
				}
				else_if = {
					limit = { tag = CES }
					MLT = { set_country_flag = mltd_ces_capitulated }
				}
				else_if = {
					limit = { tag = TBH }
					MLT = { set_country_flag = mltd_tbh_capitulated }
				}
				else_if = {
					limit = { tag = LNS }
					MLT = { set_country_flag = mltd_lns_capitulated }
				}
			}
		}
	}
}
