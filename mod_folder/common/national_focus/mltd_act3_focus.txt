# Terms for the Citadel (round 28): the Brotherhood in Washington is offered a seat in the Drowned
# Covenant (event mltd.32, which it will most likely refuse). Mutually exclusive with The Tide-Wall
# (round 28b): the Brotherhood is courted or fought.
shared_focus = {
	id = mltd_wbh_terms_for_the_citadel
	icon = GFX_goal_WBH_fellow_savages
	# 2026-09-24: its nation's Act II bead (The Final Ritual until then)
	prerequisite = { focus = mltd_wbh_the_drowned_knights }
	# round 28b: both ways, with The Tide-Wall's own line
	mutually_exclusive = { focus = mltd_wbh_the_tide_wall }
	# (10,23) since 2026-09-24, beside its bead; (10,26) until then
	x = -1
	y = 1
	relative_position_id = mltd_wbh_the_drowned_knights
	# twice The Tide-Wall's 30 on purpose (63 against 45 until 2026-09-25, when every focus of ours became 7, 30, 60, 120 or
	# 180 days): the offer is the slow road, and a refusal leaves only a war goal that expires
	cost = 60
	available = {
		# the Brotherhood that would be asked - WBH unless it is gone or someone's subject, else TCA as the Northwestern
		# Brotherhood - at peace with us, no one's subject, and in no faction or leading its own
		# (2026-09-25: a yes brings that faction over), so that it can join ours
		custom_trigger_tooltip = {
			tooltip = mltd_wbh_terms_brotherhood_tt
			any_country = { mltd_wbh_terms_target = yes }
		}
		OR = {
			is_in_faction = no
			is_faction_leader = yes
		}
		# never beside the Broken Coast: once the raiders have sworn to the Covenant, the
		# Brotherhood will not share a table with them (mltd_brk_join_the_covenant sets the flag). A
		# refused Covenant leaves this open.
		custom_trigger_tooltip = {
			tooltip = mltd_wbh_terms_no_covenant_tt
			NOT = { has_country_flag = mltd_brk_covenant_joined }
		}
		# round 28c: the same set the other two invitations ask - a cult at 40 %, the civilian
		# infiltration token and a quarter of the country covered - against whichever Brotherhood
		# would be asked. One branch per tag, the house idiom for this pair (The Drowned Knights and
		# The Tide-Wall), since the target is decided at completion
		if = {
			limit = { has_dlc = "La Resistance" }
			custom_trigger_tooltip = {
				tooltip = mltd_wbh_terms_faithful_tt
				OR = {
					AND = {
						WBH = { mltd_wbh_terms_target = yes }
						WBH = { check_variable = { mltd_cult_strength > 39.9 } }
						has_operation_token = { tag = WBH token = token_civilian }
						network_national_coverage = { target = WBH value > 0.25 }
					}
					AND = {
						TCA = { mltd_wbh_terms_target = yes }
						TCA = { check_variable = { mltd_cult_strength > 39.9 } }
						has_operation_token = { tag = TCA token = token_civilian }
						network_national_coverage = { target = TCA value > 0.25 }
					}
				}
			}
		}
		# an AI offers only when the answer can be yes (not on historical focuses), only to the Washington Brotherhood itself
		# (not the Northwestern one, which accepts a tenth of the time), not while it courts the Broken Coast, and only when The
		# Tide-Wall's war could be fought - Washington's stage, the army ready, a safe target (from
		# 2281 only while no Act III focus is done) - since a refusal hands it a war goal
		# (mltd_ai_triggers.txt, section W)
		mltd_ai_may_offer_the_citadel = yes
	}
	# 2026-09-23: bypassed once no Brotherhood holds Washington - with its nation gone it paid only
	# its fallback
	bypass = {
		custom_trigger_tooltip = {
			tooltip = mltd_wbh_no_brotherhood_tt
			NOT = { any_country = { mltd_wbh_brotherhood = yes } }
		}
	}
	cancel_if_invalid = no
	continue_if_invalid = yes
	# no ai_will_do: the default weight of 1, below the war focuses' 3, so an AI running no plan
	# prefers them; an AI's real gate is the hold above
	completion_reward = {
		# the target, MLT's own faction and the Covenant flag are tested again at completion (mltd_wbh_terms_can_offer), because
		# the world can change while this focus runs (continue_if_invalid) - the Broken Coast can accept the Covenant, MLT can
		# join someone else's faction - and an offer that cannot be accepted must not be sent; one offer, to one Brotherhood
		# (mltd_wbh_terms_target picks WBH unless it is gone or someone's subject). 2026-09-24 (the
		# user: too much text): the offer, then "If it accepts:" and "If it refuses:" over the
		# answer's own native lines - the faction, the join, the war goal - and the claims it drops.
		# 2026-09-25 (the user: it showed only the +50): the preview is an if of its own, which
		# shows the offer unless the focus is running and the offer can no longer go - the one time
		# the +50 is due. Read off the real if / else it showed only the +50 whenever no Brotherhood
		# could be asked, and with WBH leading the Northern League from 2275 that is most of the
		# game. The real if / else is hidden below it. Since the same day a Brotherhood that leads a
		# faction may be asked, and a yes brings the faction's members over with it
		if = {
			limit = {
				OR = {
					mltd_wbh_terms_can_offer = yes
					NOT = { focus_progress = { focus = mltd_wbh_terms_for_the_citadel progress > 0 } }
				}
			}
			custom_effect_tooltip = mltd_wbh_terms_offer_tt
			custom_effect_tooltip = mltd_wbh_terms_if_accepted_tt
			effect_tooltip = {
				if = {
					limit = { is_in_faction = no }
					create_faction = mltd_drowned_covenant
				}
				# the Brotherhood that would be asked, by mltd_wbh_terms_target's own rule, whatever
				# its faction or war
				if = {
					limit = {
						WBH = {
							exists = yes
							is_subject = no
						}
					}
					add_to_faction = WBH
				}
				else = { add_to_faction = TCA }
			}
			custom_effect_tooltip = mltd_wbh_terms_claims_tt
			# 2026-09-25: a Brotherhood that leads a faction - WBH leads OWB's Northern League from
			# 2275 - brings its members over (mltd_wbh_join_the_covenant); the faction's name
			# prints, its members show on the faction screen
			if = {
				limit = {
					WBH = {
						exists = yes
						is_subject = no
						is_faction_leader = yes
					}
				}
				custom_effect_tooltip = mltd_wbh_terms_faction_wbh_tt
			}
			else_if = {
				limit = {
					NOT = {
						WBH = {
							exists = yes
							is_subject = no
						}
					}
					TCA = {
						exists = yes
						is_faction_leader = yes
					}
				}
				custom_effect_tooltip = mltd_wbh_terms_faction_tca_tt
			}
			custom_effect_tooltip = mltd_wbh_terms_if_refused_tt
			effect_tooltip = {
				if = {
					limit = {
						WBH = {
							exists = yes
							is_subject = no
						}
					}
					create_wargoal = {
						type = annex_everything
						target = WBH
						expire = 365
					}
				}
				else = {
					create_wargoal = {
						type = annex_everything
						target = TCA
						expire = 365
					}
				}
			}
		}
		else = {
			effect_tooltip = { add_political_power = 50 }
		}
		hidden_effect = {
			if = {
				limit = { mltd_wbh_terms_can_offer = yes }
				random_country = {
					limit = { mltd_wbh_terms_target = yes }
					country_event = { id = mltd.32 days = 1 }
				}
			}
			else = { add_political_power = 50 }
		}
	}
}
