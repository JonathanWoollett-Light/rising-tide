# ---- Round 21: the Drowned Covenant. mltd.21 goes to the Broken Coast from MLT's focus mltd_brk_the_drowned_covenant
# (FROM = MLT); its answer reaches MLT as mltd.22 (accepted) or mltd.23 (refused), where FROM = BRK.
country_event = {
	immediate = { log = "[GetDateText]: [Root.GetName]: event mltd.21" }
	id = mltd.21
	title = mltd.21.t
	desc = mltd.21.d
	picture = GFX_event_BRK_raider_port

	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = mltd.21.a
		# Round 28: offered only when the raiders can actually join - free, at peace with MLT, and MLT free or leading its own
		# faction without the Brotherhood in Washington in it (mltd_wbh_alliance; the focus tested all of this when it sent
		# the offer, but a human may take days to answer). Otherwise only the refusal is left.
		trigger = {
			is_subject = no
			is_in_faction = no
			NOT = { has_war_with = FROM }
			FROM = {
				OR = {
					is_in_faction = no
					is_faction_leader = yes
				}
				NOT = { has_country_flag = mltd_wbh_alliance }
			}
		}
		# Round 28: a historical Broken Coast always swears (the refusal's weight drops to 0 below); an unhistorical one swears
		# half the time. The weights are relative: 50 against 50, or 50 against 0. OWB's idiom, nevada_pact_events.txt:15-22.
		# The odds are also in the focus's description (GetMltdBrkOdds); change both together.
		ai_chance = { base = 50 }
		FROM = { mltd_brk_join_the_covenant = yes }
		hidden_effect = {
			FROM = { country_event = { id = mltd.22 days = 1 } }
		}
	}
	option = {
		name = mltd.21.b
		ai_chance = {
			base = 50
			modifier = {
				is_historical_focus_on = yes
				factor = 0
			}
		}
		FROM = {
			add_opinion_modifier = {
				target = ROOT
				modifier = mltd_drowned_covenant_refused
			}
		}
		# Round 28: a refusal leaves MLT only a hunting licence - an annex_everything war goal on the raiders that expires after
		# 365 days (vanilla's expire is in days: bulgaria.txt:8961-8966, with its own expiry tooltip). ROOT is BRK here.
		# Since round 28c the Covenant and its war twin, The Coast Goes Under, exclude each other both ways, so after a refusal
		# this war goal is the tree's one road to war with the raiders (as mltd.24's is with the Bone Dancers).
		if = {
			limit = { NOT = { has_war_with = FROM } }
			FROM = {
				create_wargoal = {
					type = annex_everything
					target = ROOT
					expire = 365
				}
			}
			custom_effect_tooltip = mltd_refusal_wargoal_expires_tt
		}
		hidden_effect = {
			set_global_flag = mltd_brk_refused_covenant
			FROM = { country_event = { id = mltd.23 days = 1 } }
		}
	}
}
